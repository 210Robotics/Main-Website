39static inline uint8x8_t ALPHA_BLEND(uint8x8_t c, uint8x8_t a)
40{
41 uint16x8_t t = vmull_u8(c, a);
42 return vshrn_n_u16(t, 8);
43}

46static void neonRasterGrayscale8(uint8_t* dst, uint8_t val, uint32_t offset, int32_t len)
47{
48 dst += offset;
49
50 int32_t i = 0;
51 const uint8x16_t valVec = vdupq_n_u8(val);
52#if TVG_AARCH64
53 uint8x16x4_t valQuad = {valVec, valVec, valVec, valVec};
54 for (; i <= len - 16 * 4; i += 16 * 4) {
55 vst1q_u8_x4(dst + i, valQuad);
56 }
57#else
58 for (; i <= len - 16; i += 16) {
59 vst1q_u8(dst + i, valVec);
60 }
61#endif
62 for (; i < len; i++) {
63 dst[i] = val;
64 }
65}

68static void neonRasterPixel32(uint32_t *dst, uint32_t val, uint32_t offset, int32_t len)
69{
70 dst += offset;
71
72 uint32x4_t vectorVal = vdupq_n_u32(val);
73
74#if TVG_AARCH64
75 uint32_t iterations = len / 16;
76 uint32_t neonFilled = iterations * 16;
77 uint32x4x4_t valQuad = {vectorVal, vectorVal, vectorVal, vectorVal};
78 for (uint32_t i = 0; i < iterations; ++i) {
79 vst4q_u32(dst, valQuad);
80 dst += 16;
81 }
82#else
83 uint32_t iterations = len / 4;
84 uint32_t neonFilled = iterations * 4;
85 for (uint32_t i = 0; i < iterations; ++i) {
86 vst1q_u32(dst, vectorVal);
87 dst += 4;
88 }
89#endif
90 int32_t leftovers = len - neonFilled;
91 while (leftovers--) *dst++ = val;
92}

95static bool neonRasterTranslucentRle(SwSurface* surface, const SwRleData* rle, uint8_t r, uint8_t g, uint8_t b, uint8_t a)
96{
97 if (surface->channelSize != sizeof(uint32_t)) {
98 TVGERR("SW_ENGINE", "Unsupported Channel Size = %d", surface->channelSize);
99 return false;
100 }
101
102 auto color = surface->join(r, g, b, a);
103 auto span = rle->spans;
104 uint32_t src;
105 uint8x8_t *vDst = nullptr;
106 uint16_t align;
107
108 for (uint32_t i = 0; i < rle->size; ++i) {
109 if (span->coverage < 255) src = ALPHA_BLEND(color, span->coverage);
110 else src = color;
111
112 auto dst = &surface->buf32[span->y * surface->stride + span->x];
113 auto ialpha = IA(src);
114
115 if ((((uintptr_t) dst) & 0x7) != 0) {
116 //fill not aligned byte
117 *dst = src + ALPHA_BLEND(*dst, ialpha);
118 vDst = (uint8x8_t*)(dst + 1);
119 align = 1;
120 } else {
121 vDst = (uint8x8_t*) dst;
122 align = 0;
123 }
124
125 uint8x8_t vSrc = (uint8x8_t) vdup_n_u32(src);
126 uint8x8_t vIalpha = vdup_n_u8((uint8_t) ialpha);
127
128 for (uint32_t x = 0; x < (span->len - align) / 2; ++x)
129 vDst[x] = vadd_u8(vSrc, ALPHA_BLEND(vDst[x], vIalpha));
130
131 auto leftovers = (span->len - align) % 2;
132 if (leftovers > 0) dst[span->len - 1] = src + ALPHA_BLEND(dst[span->len - 1], ialpha);
133
134 ++span;
135 }
136 return true;
137}

140static bool neonRasterTranslucentRect(SwSurface* surface, const SwBBox& region, uint8_t r, uint8_t g, uint8_t b, uint8_t a)
141{
142 if (surface->channelSize != sizeof(uint32_t)) {
143 TVGERR("SW_ENGINE", "Unsupported Channel Size = %d", surface->channelSize);
144 return false;
145 }
146
147 auto color = surface->join(r, g, b, a);
148 auto buffer = surface->buf32 + (region.min.y * surface->stride) + region.min.x;
149 auto h = static_cast<uint32_t>(region.max.y - region.min.y);
150 auto w = static_cast<uint32_t>(region.max.x - region.min.x);
151 auto ialpha = 255 - a;
152
153 auto vColor = vdup_n_u32(color);
154 auto vIalpha = vdup_n_u8((uint8_t) ialpha);
155
156 uint8x8_t* vDst = nullptr;
157 uint32_t align;
158
159 for (uint32_t y = 0; y < h; ++y) {
160 auto dst = &buffer[y * surface->stride];
161
162 if ((((uintptr_t) dst) & 0x7) != 0) {
163 //fill not aligned byte
164 *dst = color + ALPHA_BLEND(*dst, ialpha);
165 vDst = (uint8x8_t*) (dst + 1);
166 align = 1;
167 } else {
168 vDst = (uint8x8_t*) dst;
169 align = 0;
170 }
171
172 for (uint32_t x = 0; x < (w - align) / 2; ++x)
173 vDst[x] = vadd_u8((uint8x8_t)vColor, ALPHA_BLEND(vDst[x], vIalpha));
174
175 auto leftovers = (w - align) % 2;
176 if (leftovers > 0) dst[w - 1] = color + ALPHA_BLEND(dst[w - 1], ialpha);
177 }
178 return true;
179}
