34/**
35 * \enum AivisionDetectType
36 * \ingroup cpp-aivision
37 * Enum class for describing detection type of objects detected by the AI Vision Sensor.
38 */
39enum class AivisionDetectType : uint8_t {
40 color = (1 << 0), /**< object was detected based on color descriptor */
41 code = (1 << 1), /**< object was detected based on code descriptor */
42 object = (1 << 2), /**< object was detected using AI model */
43 tag = (1 << 3) /**< object was detected as an AprilTag */
44};
45
46/**
47 * \enum AivisionModeType
48 * \ingroup cpp-aivision
49 * Enum class for enabling/disabling detection types of AI Vision Sensor.
50 */
51enum class AivisionModeType : uint8_t {
52 tags = (1 << 0), /**< AprilTag detection */
53 colors = (1 << 1), /**< color and code detection */
54 objects = (1 << 2), /**< AI model object detection */
55 color_merge = (1 << 4), /**< merge adjacent color detections */
56 all = (1 << 0) | (1 << 1) | (1 << 2),
57};

64enum class AivisionTagFamily { tag_21H7 = 0, tag_16H5 = 1, tag_25H9 = 2, tag_61H11 = 3 };
