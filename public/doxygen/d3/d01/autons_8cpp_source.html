327// Calculate the offsets of your tracking wheels
328///
329void measure_offsets() {
330 // Number of times to test
331 int iterations = 10;
332
333 // Our final offsets
334 double l_offset = 0.0, r_offset = 0.0, b_offset = 0.0, f_offset = 0.0;
335
336 // Reset all trackers if they exist
337 if (chassis.odom_tracker_left != nullptr) chassis.odom_tracker_left->reset();
338 if (chassis.odom_tracker_right != nullptr) chassis.odom_tracker_right->reset();
339 if (chassis.odom_tracker_back != nullptr) chassis.odom_tracker_back->reset();
340 if (chassis.odom_tracker_front != nullptr) chassis.odom_tracker_front->reset();
341
342 for (int i = 0; i < iterations; i++) {
343 // Reset pid targets and get ready for running an auton
344 chassis.pid_targets_reset();
345 chassis.drive_imu_reset();
346 chassis.drive_sensor_reset();
347 chassis.drive_brake_set(pros::E_MOTOR_BRAKE_HOLD);
348 chassis.odom_xyt_set(0_in, 0_in, 0_deg);
349 double imu_start = chassis.odom_theta_get();
350 double target = i % 2 == 0 ? 90 : 270; // Switch the turn target every run from 270 to 90
351
352 // Turn to target at half power
353 chassis.pid_turn_set(target, 63, ez::raw);
354 chassis.pid_wait();
355 pros::delay(250);
356
357 // Calculate delta in angle
358 double t_delta = util::to_rad(fabs(util::wrap_angle(chassis.odom_theta_get() - imu_start)));
359
360 // Calculate delta in sensor values that exist
361 double l_delta = chassis.odom_tracker_left != nullptr ? chassis.odom_tracker_left->get() : 0.0;
362 double r_delta = chassis.odom_tracker_right != nullptr ? chassis.odom_tracker_right->get() : 0.0;
363 double b_delta = chassis.odom_tracker_back != nullptr ? chassis.odom_tracker_back->get() : 0.0;
364 double f_delta = chassis.odom_tracker_front != nullptr ? chassis.odom_tracker_front->get() : 0.0;
365
366 // Calculate the radius that the robot traveled
367 l_offset += l_delta / t_delta;
368 r_offset += r_delta / t_delta;
369 b_offset += b_delta / t_delta;
370 f_offset += f_delta / t_delta;
371 }
372
373 // Average all offsets
374 l_offset /= iterations;
375 r_offset /= iterations;
376 b_offset /= iterations;
377 f_offset /= iterations;
378
379 // Set new offsets to trackers that exist
380 if (chassis.odom_tracker_left != nullptr) chassis.odom_tracker_left->distance_to_center_set(l_offset);
381 if (chassis.odom_tracker_right != nullptr) chassis.odom_tracker_right->distance_to_center_set(r_offset);
382 if (chassis.odom_tracker_back != nullptr) chassis.odom_tracker_back->distance_to_center_set(b_offset);
383 if (chassis.odom_tracker_front != nullptr) chassis.odom_tracker_front->distance_to_center_set(f_offset);
384
385 pros::lcd::print(0, "L Offset: %f", l_offset);
386 pros::lcd::print(1, "R Offset: %f", r_offset);
387}
