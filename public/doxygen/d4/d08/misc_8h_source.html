271/**
272 * \def Given an id and a port, this macro sets the port variable based on the id and allows the mutex to take that
273 * port.
274 *
275 * \returns error (in the function/scope it's in) if the controller failed to connect or an invalid id is given.
276 */
277#define CONTROLLER_PORT_MUTEX_TAKE(id, port)
278 switch (id) {
279 case E_CONTROLLER_MASTER:
280 port = V5_PORT_CONTROLLER_1;
281 break;
282 case E_CONTROLLER_PARTNER:
283 port = V5_PORT_CONTROLLER_2;
284 break;
285 default:
286 errno = EINVAL;
287 return PROS_ERR;
288 }
289 if (!internal_port_mutex_take(port)) {
290 errno = EACCES;
291 return PROS_ERR;
292 }
