295/*
296 * Approximate sqrt(x*x+y*y) using the `alpha max plus beta min'
297 * algorithm. We use alpha = 1, beta = 3/8, giving us results with a
298 * largest error less than 7% compared to the exact value.
299 */
300static inline SwCoord HYPOT(SwPoint pt)
301{
302 if (pt.x < 0) pt.x = -pt.x;
303 if (pt.y < 0) pt.y = -pt.y;
304 return ((pt.x > pt.y) ? (pt.x + (3 * pt.y >> 3)) : (pt.y + (3 * pt.x >> 3)));
305}

307static void _genSpan(SwRleData* rle, const SwSpan* spans, uint32_t count)
308{
309 auto newSize = rle->size + count;
310
311 /* allocate enough memory for new spans */
312 /* alloc is required to prevent free and reallocation */
313 /* when the rle needs to be regenerated because of attribute change. */
314 if (rle->alloc < newSize) {
315 rle->alloc = (newSize * 2);
316 //OPTIMIZE: use mempool!
317 rle->spans = static_cast<SwSpan*>(realloc(rle->spans, rle->alloc * sizeof(SwSpan)));
318 }
319
320 //copy the new spans to the allocated memory
321 SwSpan* lastSpan = rle->spans + rle->size;
322 memcpy(lastSpan, spans, count * sizeof(SwSpan));
323
324 rle->size = newSize;
325}

328static void _horizLine(RleWorker& rw, SwCoord x, SwCoord y, SwCoord area, SwCoord aCount)
329{
330 x += rw.cellMin.x;
331 y += rw.cellMin.y;
332
333 //Clip Y range
334 if (y < rw.cellMin.y || y >= rw.cellMax.y) return;
335
336 /* compute the coverage line's coverage, depending on the outline fill rule */
337 /* the coverage percentage is area/(PIXEL_BITS*PIXEL_BITS*2) */
338 auto coverage = static_cast<int>(area >> (PIXEL_BITS * 2 + 1 - 8)); //range 0 - 255
339
340 if (coverage < 0) coverage = -coverage;
341
342 if (rw.outline->fillRule == FillRule::EvenOdd) {
343 coverage &= 511;
344 if (coverage > 255) coverage = 511 - coverage;
345 } else {
346 //normal non-zero winding rule
347 if (coverage > 255) coverage = 255;
348 }
349
350 //span has ushort coordinates. check limit overflow
351 if (x >= SHRT_MAX) {
352 TVGERR("SW_ENGINE", "X-coordinate overflow!");
353 x = SHRT_MAX;
354 }
355 if (y >= SHRT_MAX) {
356 TVGERR("SW_ENGINE", "Y Coordinate overflow!");
357 y = SHRT_MAX;
358 }
359
360 if (coverage > 0) {
361 if (!rw.antiAlias) coverage = 255;
362 auto count = rw.spansCnt;
363 auto span = rw.spans + count - 1;
364
365 //see whether we can add this span to the current list
366 if ((count > 0) && (rw.ySpan == y) &&
367 (span->x + span->len == x) && (span->coverage == coverage)) {
368
369 //Clip x range
370 SwCoord xOver = 0;
371 if (x + aCount >= rw.cellMax.x) xOver -= (x + aCount - rw.cellMax.x);
372 if (x < rw.cellMin.x) xOver -= (rw.cellMin.x - x);
373
374 //span->len += (aCount + xOver) - 1;
375 span->len += (aCount + xOver);
376 return;
377 }
378
379 if (count >= MAX_SPANS) {
380 _genSpan(rw.rle, rw.spans, count);
381 rw.spansCnt = 0;
382 rw.ySpan = 0;
383 span = rw.spans;
384 } else {
385 ++span;
386 }
387
388 //Clip x range
389 SwCoord xOver = 0;
390 if (x + aCount >= rw.cellMax.x) xOver -= (x + aCount - rw.cellMax.x);
391 if (x < rw.cellMin.x) {
392 xOver -= (rw.cellMin.x - x);
393 x = rw.cellMin.x;
394 }
395
396 //Nothing to draw
397 if (aCount + xOver <= 0) return;
398
399 //add a span to the current list
400 span->x = x;
401 span->y = y;
402 span->len = (aCount + xOver);
403 span->coverage = coverage;
404 ++rw.spansCnt;
405 rw.ySpan = y;
406 }
407}

410static void _sweep(RleWorker& rw)
411{
412 if (rw.cellsCnt == 0) return;
413
414 rw.spansCnt = 0;
415 rw.ySpan = 0;
416
417 for (int y = 0; y < rw.yCnt; ++y) {
418 auto cover = 0;
419 auto x = 0;
420 auto cell = rw.yCells[y];
421
422 while (cell) {
423 if (cell->x > x && cover != 0) _horizLine(rw, x, y, cover * (ONE_PIXEL * 2), cell->x - x);
424 cover += cell->cover;
425 auto area = cover * (ONE_PIXEL * 2) - cell->area;
426 if (area != 0 && cell->x >= 0) _horizLine(rw, cell->x, y, area, 1);
427 x = cell->x + 1;
428 cell = cell->next;
429 }
430
431 if (cover != 0) _horizLine(rw, x, y, cover * (ONE_PIXEL * 2), rw.cellXCnt - x);
432 }
433
434 if (rw.spansCnt > 0) _genSpan(rw.rle, rw.spans, rw.spansCnt);
435}

438static Cell* _findCell(RleWorker& rw)
439{
440 auto x = rw.cellPos.x;
441 if (x > rw.cellXCnt) x = rw.cellXCnt;
442
443 auto pcell = &rw.yCells[rw.cellPos.y];
444
445 while(true) {
446 Cell* cell = *pcell;
447 if (!cell || cell->x > x) break;
448 if (cell->x == x) return cell;
449 pcell = &cell->next;
450 }
451
452 if (rw.cellsCnt >= rw.maxCells) longjmp(rw.jmpBuf, 1);
453
454 auto cell = rw.cells + rw.cellsCnt++;
455 cell->x = x;
456 cell->area = 0;
457 cell->cover = 0;
458 cell->next = *pcell;
459 *pcell = cell;
460
461 return cell;
462}

465static void _recordCell(RleWorker& rw)
466{
467 if (rw.area | rw.cover) {
468 auto cell = _findCell(rw);
469 cell->area += rw.area;
470 cell->cover += rw.cover;
471 }

475static void _setCell(RleWorker& rw, SwPoint pos)
476{
477 /* Move the cell pointer to a new position. We set the `invalid' */
478 /* flag to indicate that the cell isn't part of those we're interested */
479 /* in during the render phase. This means that: */
480 /* */
481 /* . the new vertical position must be within min_ey..max_ey-1. */
482 /* . the new horizontal position must be strictly less than max_ex */
483 /* */
484 /* Note that if a cell is to the left of the clipping region, it is */
485 /* actually set to the (min_ex-1) horizontal position. */
486
487 /* All cells that are on the left of the clipping region go to the
488 min_ex - 1 horizontal position. */
489 pos.x -= rw.cellMin.x;
490 pos.y -= rw.cellMin.y;
491
492 if (pos.x > rw.cellMax.x) pos.x = rw.cellMax.x;
493
494 //Are we moving to a different cell?
495 if (pos != rw.cellPos) {
496 //Record the current one if it is valid
497 if (!rw.invalid) _recordCell(rw);
498 }
499
500 rw.area = 0;
501 rw.cover = 0;
502 rw.cellPos = pos;
503 rw.invalid = ((unsigned)pos.y >= (unsigned)rw.cellYCnt || pos.x >= rw.cellXCnt);
504}

507static void _startCell(RleWorker& rw, SwPoint pos)
508{
509 if (pos.x > rw.cellMax.x) pos.x = rw.cellMax.x;
510 if (pos.x < rw.cellMin.x) pos.x = rw.cellMin.x;
511
512 rw.area = 0;
513 rw.cover = 0;
514 rw.cellPos = pos - rw.cellMin;
515 rw.invalid = false;
516
517 _setCell(rw, pos);
518}

521static void _moveTo(RleWorker& rw, const SwPoint& to)
522{
523 //record current cell, if any */
524 if (!rw.invalid) _recordCell(rw);
525
526 //start to a new position
527 _startCell(rw, TRUNC(to));
528
529 rw.pos = to;
530}

533static void _lineTo(RleWorker& rw, const SwPoint& to)
534{
535#define SW_UDIV(a, b)
536 static_cast<SwCoord>(((unsigned long)(a) * (unsigned long)(b)) >>
537 (sizeof(long) * CHAR_BIT - PIXEL_BITS))
538
539 auto e1 = TRUNC(rw.pos);
540 auto e2 = TRUNC(to);
541
542 //vertical clipping
543 if ((e1.y >= rw.cellMax.y && e2.y >= rw.cellMax.y) || (e1.y < rw.cellMin.y && e2.y < rw.cellMin.y)) {
544 rw.pos = to;
545 return;
546 }
547
548 auto diff = to - rw.pos;
549 auto f1 = rw.pos - SUBPIXELS(e1);
550 SwPoint f2;
551
552 //inside one cell
553 if (e1 == e2) {
554 ;
555 //any horizontal line
556 } else if (diff.y == 0) {
557 e1.x = e2.x;
558 _setCell(rw, e1);
559 } else if (diff.x == 0) {
560 //vertical line up
561 if (diff.y > 0) {
562 do {
563 f2.y = ONE_PIXEL;
564 rw.cover += (f2.y - f1.y);
565 rw.area += (f2.y - f1.y) * f1.x * 2;
566 f1.y = 0;
567 ++e1.y;
568 _setCell(rw, e1);
569 } while(e1.y != e2.y);
570 //vertical line down
571 } else {
572 do {
573 f2.y = 0;
574 rw.cover += (f2.y - f1.y);
575 rw.area += (f2.y - f1.y) * f1.x * 2;
576 f1.y = ONE_PIXEL;
577 --e1.y;
578 _setCell(rw, e1);
579 } while(e1.y != e2.y);
580 }
581 //any other line
582 } else {
583 Area prod = diff.x * f1.y - diff.y * f1.x;
584
585 /* These macros speed up repetitive divisions by replacing them
586 with multiplications and right shifts. */
587 auto dx_r = static_cast<long>(ULONG_MAX >> PIXEL_BITS) / (diff.x);
588 auto dy_r = static_cast<long>(ULONG_MAX >> PIXEL_BITS) / (diff.y);
589
590 /* The fundamental value `prod' determines which side and the */
591 /* exact coordinate where the line exits current cell. It is */
592 /* also easily updated when moving from one cell to the next. */
593 do {
594 auto px = diff.x * ONE_PIXEL;
595 auto py = diff.y * ONE_PIXEL;
596
597 //left
598 if (prod <= 0 && prod - px > 0) {
599 f2 = {0, SW_UDIV(-prod, -dx_r)};
600 prod -= py;
601 rw.cover += (f2.y - f1.y);
602 rw.area += (f2.y - f1.y) * (f1.x + f2.x);
603 f1 = {ONE_PIXEL, f2.y};
604 --e1.x;
605 //up
606 } else if (prod - px <= 0 && prod - px + py > 0) {
607 prod -= px;
608 f2 = {SW_UDIV(-prod, dy_r), ONE_PIXEL};
609 rw.cover += (f2.y - f1.y);
610 rw.area += (f2.y - f1.y) * (f1.x + f2.x);
611 f1 = {f2.x, 0};
612 ++e1.y;
613 //right
614 } else if (prod - px + py <= 0 && prod + py >= 0) {
615 prod += py;
616 f2 = {ONE_PIXEL, SW_UDIV(prod, dx_r)};
617 rw.cover += (f2.y - f1.y);
618 rw.area += (f2.y - f1.y) * (f1.x + f2.x);
619 f1 = {0, f2.y};
620 ++e1.x;
621 //down
622 } else {
623 f2 = {SW_UDIV(prod, -dy_r), 0};
624 prod += px;
625 rw.cover += (f2.y - f1.y);
626 rw.area += (f2.y - f1.y) * (f1.x + f2.x);
627 f1 = {f2.x, ONE_PIXEL};
628 --e1.y;
629 }
630
631 _setCell(rw, e1);
632
633 } while(e1 != e2);
634 }
635
636 f2 = {to.x - SUBPIXELS(e2.x), to.y - SUBPIXELS(e2.y)};
637 rw.cover += (f2.y - f1.y);
638 rw.area += (f2.y - f1.y) * (f1.x + f2.x);
639 rw.pos = to;
640}

643static void _cubicTo(RleWorker& rw, const SwPoint& ctrl1, const SwPoint& ctrl2, const SwPoint& to)
644{
645 auto arc = rw.bezStack;
646 arc[0] = to;
647 arc[1] = ctrl2;
648 arc[2] = ctrl1;
649 arc[3] = rw.pos;
650
651 //Short-cut the arc that crosses the current band
652 auto min = arc[0].y;
653 auto max = arc[0].y;
654
655 SwCoord y;
656 for (auto i = 1; i < 4; ++i) {
657 y = arc[i].y;
658 if (y < min) min = y;
659 if (y > max) max = y;
660 }
661
662 if (TRUNC(min) >= rw.cellMax.y || TRUNC(max) < rw.cellMin.y) goto draw;
663
664 /* Decide whether to split or draw. See `Rapid Termination */
665 /* Evaluation for Recursive Subdivision of Bezier Curves' by Thomas */
666 /* F. Hain, at */
667 /* http://www.cis.southalabama.edu/~hain/general/Publications/Bezier/Camera-ready%20CISST02%202.pdf */
668 while (true) {
669 {
670 //diff is the P0 - P3 chord vector
671 auto diff = arc[3] - arc[0];
672 auto L = HYPOT(diff);
673
674 //avoid possible arithmetic overflow below by splitting
675 if (L > SHRT_MAX) goto split;
676
677 //max deviation may be as much as (s/L) * 3/4 (if Hain's v = 1)
678 auto sLimit = L * (ONE_PIXEL / 6);
679
680 auto diff1 = arc[1] - arc[0];
681 auto s = diff.y * diff1.x - diff.x * diff1.y;
682 if (s < 0) s = -s;
683 if (s > sLimit) goto split;
684
685 //s is L * the perpendicular distance from P2 to the line P0 - P3
686 auto diff2 = arc[2] - arc[0];
687 s = diff.y * diff2.x - diff.x * diff2.y;
688 if (s < 0) s = -s;
689 if (s > sLimit) goto split;
690
691 /* Split super curvy segments where the off points are so far
692 from the chord that the angles P0-P1-P3 or P0-P2-P3 become
693 acute as detected by appropriate dot products */
694 if (diff1.x * (diff1.x - diff.x) + diff1.y * (diff1.y - diff.y) > 0 ||
695 diff2.x * (diff2.x - diff.x) + diff2.y * (diff2.y - diff.y) > 0)
696 goto split;
697
698 //no reason to split
699 goto draw;
700 }
701 split:
702 mathSplitCubic(arc);
703 arc += 3;
704 continue;
705
706 draw:
707 _lineTo(rw, arc[0]);
708 if (arc == rw.bezStack) return;
709 arc -= 3;
710 }

714static void _decomposeOutline(RleWorker& rw)
715{
716 auto outline = rw.outline;
717 auto first = 0; //index of first point in contour
718
719 for (auto cntr = outline->cntrs.begin(); cntr < outline->cntrs.end(); ++cntr) {
720 auto last = *cntr;
721 auto limit = outline->pts.data + last;
722 auto start = UPSCALE(outline->pts[first]);
723 auto pt = outline->pts.data + first;
724 auto types = outline->types.data + first;
725
726 _moveTo(rw, UPSCALE(outline->pts[first]));
727
728 while (pt < limit) {
729 ++pt;
730 ++types;
731
732 //emit a single line_to
733 if (types[0] == SW_CURVE_TYPE_POINT) {
734 _lineTo(rw, UPSCALE(*pt));
735 //types cubic
736 } else {
737 pt += 2;
738 types += 2;
739
740 if (pt <= limit) {
741 _cubicTo(rw, UPSCALE(pt[-2]), UPSCALE(pt[-1]), UPSCALE(pt[0]));
742 continue;
743 }
744 _cubicTo(rw, UPSCALE(pt[-2]), UPSCALE(pt[-1]), start);
745 goto close;
746 }
747 }
748 _lineTo(rw, start);
749 close:
750 first = last + 1;
751 }
752}

755static int _genRle(RleWorker& rw)
756{
757 if (setjmp(rw.jmpBuf) == 0) {
758 _decomposeOutline(rw);
759 if (!rw.invalid) _recordCell(rw);
760 return 0;
761 }
762 return -1; //lack of cell memory
763}

766static SwSpan* _intersectSpansRegion(const SwRleData *clip, const SwRleData *target, SwSpan *outSpans, uint32_t outSpansCnt)
767{
768 auto out = outSpans;
769 auto spans = target->spans;
770 auto end = target->spans + target->size;
771 auto clipSpans = clip->spans;
772 auto clipEnd = clip->spans + clip->size;
773
774 while (spans < end && clipSpans < clipEnd) {
775 //align y coordinates.
776 if (clipSpans->y > spans->y) {
777 ++spans;
778 continue;
779 }
780 if (spans->y > clipSpans->y) {
781 ++clipSpans;
782 continue;
783 }
784
785 //Try clipping with all clip spans which have a same y coordinate.
786 auto temp = clipSpans;
787 while(temp < clipEnd && outSpansCnt > 0 && temp->y == clipSpans->y) {
788 auto sx1 = spans->x;
789 auto sx2 = sx1 + spans->len;
790 auto cx1 = temp->x;
791 auto cx2 = cx1 + temp->len;
792
793 //The span must be left(x1) to right(x2) direction. Not intersected.
794 if (cx2 < sx1 || sx2 < cx1) {
795 ++temp;
796 continue;
797 }
798
799 //clip span region.
800 auto x = sx1 > cx1 ? sx1 : cx1;
801 auto len = (sx2 < cx2 ? sx2 : cx2) - x;
802 if (len > 0) {
803 out->x = x;
804 out->y = temp->y;
805 out->len = len;
806 out->coverage = (uint8_t)(((spans->coverage * temp->coverage) + 0xff) >> 8);
807 ++out;
808 --outSpansCnt;
809 }
810 ++temp;
811 }
812 ++spans;
813 }
814 return out;
815}

818static SwSpan* _intersectSpansRect(const SwBBox *bbox, const SwRleData *targetRle, SwSpan *outSpans, uint32_t outSpansCnt)
819{
820 auto out = outSpans;
821 auto spans = targetRle->spans;
822 auto end = targetRle->spans + targetRle->size;
823 auto minx = static_cast<int16_t>(bbox->min.x);
824 auto miny = static_cast<int16_t>(bbox->min.y);
825 auto maxx = minx + static_cast<int16_t>(bbox->max.x - bbox->min.x) - 1;
826 auto maxy = miny + static_cast<int16_t>(bbox->max.y - bbox->min.y) - 1;
827
828 while (outSpansCnt > 0 && spans < end) {
829 if (spans->y > maxy) {
830 spans = end;
831 break;
832 }
833 if (spans->y < miny || spans->x > maxx || spans->x + spans->len <= minx) {
834 ++spans;
835 continue;
836 }
837 if (spans->x < minx) {
838 out->len = (spans->len - (minx - spans->x)) < (maxx - minx + 1) ? (spans->len - (minx - spans->x)) : (maxx - minx + 1);
839 out->x = minx;
840 }
841 else {
842 out->x = spans->x;
843 out->len = spans->len < (maxx - spans->x + 1) ? spans->len : (maxx - spans->x + 1);
844 }
845 if (out->len > 0) {
846 out->y = spans->y;
847 out->coverage = spans->coverage;
848 ++out;
849 --outSpansCnt;
850 }
851 ++spans;
852 }
853 return out;
854}

857static SwSpan* _mergeSpansRegion(const SwRleData *clip1, const SwRleData *clip2, SwSpan *outSpans)
858{
859 auto out = outSpans;
860 auto spans1 = clip1->spans;
861 auto end1 = clip1->spans + clip1->size;
862 auto spans2 = clip2->spans;
863 auto end2 = clip2->spans + clip2->size;
864
865 //list two spans up in y order
866 //TODO: Remove duplicated regions?
867 while (spans1 < end1 && spans2 < end2) {
868 while (spans1 < end1 && spans1->y <= spans2->y) {
869 *out = *spans1;
870 ++spans1;
871 ++out;
872 }
873 if (spans1 >= end1) break;
874 while (spans2 < end2 && spans2->y <= spans1->y) {
875 *out = *spans2;
876 ++spans2;
877 ++out;
878 }
879 }
880
881 //Leftovers
882 while (spans1 < end1) {
883 *out = *spans1;
884 ++spans1;
885 ++out;
886 }
887 while (spans2 < end2) {
888 *out = *spans2;
889 ++spans2;
890 ++out;
891 }
892
893 return out;
894}

909SwRleData* rleRender(SwRleData* rle, const SwOutline* outline, const SwBBox& renderRegion, bool antiAlias)
910{
911 constexpr auto RENDER_POOL_SIZE = 16384L;
912 constexpr auto BAND_SIZE = 40;
913
914 //TODO: We can preserve several static workers in advance
915 RleWorker rw;
916 Cell buffer[RENDER_POOL_SIZE / sizeof(Cell)];
917
918 //Init Cells
919 rw.buffer = buffer;
920 rw.bufferSize = sizeof(buffer);
921 rw.yCells = reinterpret_cast<Cell**>(buffer);
922 rw.cells = nullptr;
923 rw.maxCells = 0;
924 rw.cellsCnt = 0;
925 rw.area = 0;
926 rw.cover = 0;
927 rw.invalid = true;
928 rw.cellMin = renderRegion.min;
929 rw.cellMax = renderRegion.max;
930 rw.cellXCnt = rw.cellMax.x - rw.cellMin.x;
931 rw.cellYCnt = rw.cellMax.y - rw.cellMin.y;
932 rw.ySpan = 0;
933 rw.outline = const_cast<SwOutline*>(outline);
934 rw.bandSize = rw.bufferSize / (sizeof(Cell) * 8); //bandSize: 64
935 rw.bandShoot = 0;
936 rw.antiAlias = antiAlias;
937
938 if (!rle) rw.rle = reinterpret_cast<SwRleData*>(calloc(1, sizeof(SwRleData)));
939 else rw.rle = rle;
940
941 //Generate RLE
942 Band bands[BAND_SIZE];
943 Band* band;
944
945 /* set up vertical bands */
946 auto bandCnt = static_cast<int>((rw.cellMax.y - rw.cellMin.y) / rw.bandSize);
947 if (bandCnt == 0) bandCnt = 1;
948 else if (bandCnt >= BAND_SIZE) bandCnt = (BAND_SIZE - 1);
949
950 auto min = rw.cellMin.y;
951 auto yMax = rw.cellMax.y;
952 SwCoord max;
953 int ret;
954
955 for (int n = 0; n < bandCnt; ++n, min = max) {
956 max = min + rw.bandSize;
957 if (n == bandCnt -1 || max > yMax) max = yMax;
958
959 bands[0].min = min;
960 bands[0].max = max;
961 band = bands;
962
963 while (band >= bands) {
964 rw.yCells = static_cast<Cell**>(rw.buffer);
965 rw.yCnt = band->max - band->min;
966
967 int cellStart = sizeof(Cell*) * (int)rw.yCnt;
968 int cellMod = cellStart % sizeof(Cell);
969
970 if (cellMod > 0) cellStart += sizeof(Cell) - cellMod;
971
972 auto cellEnd = rw.bufferSize;
973 cellEnd -= cellEnd % sizeof(Cell);
974
975 auto cellsMax = reinterpret_cast<Cell*>((char*)rw.buffer + cellEnd);
976 rw.cells = reinterpret_cast<Cell*>((char*)rw.buffer + cellStart);
977
978 if (rw.cells >= cellsMax) goto reduce_bands;
979
980 rw.maxCells = cellsMax - rw.cells;
981 if (rw.maxCells < 2) goto reduce_bands;
982
983 for (int y = 0; y < rw.yCnt; ++y)
984 rw.yCells[y] = nullptr;
985
986 rw.cellsCnt = 0;
987 rw.invalid = true;
988 rw.cellMin.y = band->min;
989 rw.cellMax.y = band->max;
990 rw.cellYCnt = band->max - band->min;
991
992 ret = _genRle(rw);
993 if (ret == 0) {
994 _sweep(rw);
995 --band;
996 continue;
997 } else if (ret == 1) {
998 goto error;
999 }
1000
1001 reduce_bands:
1002 /* render pool overflow: we will reduce the render band by half */
1003 auto bottom = band->min;
1004 auto top = band->max;
1005 auto middle = bottom + ((top - bottom) >> 1);
1006
1007 /* This is too complex for a single scanline; there must
1008 be some problems */
1009 if (middle == bottom) goto error;
1010
1011 if (bottom - top >= rw.bandSize) ++rw.bandShoot;
1012
1013 band[1].min = bottom;
1014 band[1].max = middle;
1015 band[0].min = middle;
1016 band[0].max = top;
1017 ++band;
1018 }
1019 }
1020
1021 if (rw.bandShoot > 8 && rw.bandSize > 16)
1022 rw.bandSize = (rw.bandSize >> 1);
1023
1024 return rw.rle;
1025
1026error:
1027 free(rw.rle);
1028 rw.rle = nullptr;
1029 return nullptr;
1030}

1071{
1072 if (!rle || (!clip1 && !clip2)) return;
1073 if (clip1 && clip1->size == 0 && clip2 && clip2->size == 0) return;
1074
1075 TVGLOG("SW_ENGINE", "Unifying Rle!");
1076
1077 //clip1 is empty, just copy clip2
1078 if (!clip1 || clip1->size == 0) {
1079 if (clip2) {
1080 auto spans = static_cast<SwSpan*>(malloc(sizeof(SwSpan) * (clip2->size)));
1081 memcpy(spans, clip2->spans, clip2->size);
1082 _replaceClipSpan(rle, spans, clip2->size);
1083 } else {
1084 _replaceClipSpan(rle, nullptr, 0);
1085 }
1086 return;
1087 }
1088
1089 //clip2 is empty, just copy clip1
1090 if (!clip2 || clip2->size == 0) {
1091 if (clip1) {
1092 auto spans = static_cast<SwSpan*>(malloc(sizeof(SwSpan) * (clip1->size)));
1093 memcpy(spans, clip1->spans, clip1->size);
1094 _replaceClipSpan(rle, spans, clip1->size);
1095 } else {
1096 _replaceClipSpan(rle, nullptr, 0);
1097 }
1098 return;
1099 }
1100
1101 auto spanCnt = clip1->size + clip2->size;
1102 auto spans = static_cast<SwSpan*>(malloc(sizeof(SwSpan) * spanCnt));
1103 auto spansEnd = _mergeSpansRegion(clip1, clip2, spans);
1104
1105 _replaceClipSpan(rle, spans, spansEnd - spans);
1106}
