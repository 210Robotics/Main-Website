25/*All possible property value types*/

27#define LV_PROPERTY_TYPE_INT 1 /*int32_t type*/
28#define LV_PROPERTY_TYPE_PRECISE 2 /*lv_value_precise_t, int32_t or float depending on LV_USE_FLOAT*/
29#define LV_PROPERTY_TYPE_COLOR 3 /*ARGB8888 type*/
30#define LV_PROPERTY_TYPE_POINT 4 /*lv_point_t */
31#define LV_PROPERTY_TYPE_POINTER 5 /*void * pointer*/
32#define LV_PROPERTY_TYPE_IMGSRC 6 /*Special pointer for image*/
33#define LV_PROPERTY_TYPE_TEXT 7 /*Special pointer of char* */
34#define LV_PROPERTY_TYPE_OBJ 8 /*Special pointer of lv_obj_t* */
35#define LV_PROPERTY_TYPE_DISPLAY 9 /*Special pointer of lv_display_t* */

39#define LV_PROPERTY_TYPE_SHIFT 28

83typedef struct {
84 lv_prop_id_t id;
85 union {
86 int32_t num; /**< Number integer number (opacity, enums, booleans or "normal" numbers)*/
87 bool enable; /**< booleans*/
88 const void * ptr; /**< Constant pointers (font, cone text, etc)*/
89 lv_color_t color; /**< Colors*/
90 lv_value_precise_t precise; /**< float or int for precise value*/
91 lv_point_t point; /**< Point*/
92 struct {
93 /**
94 * Note that place struct member `style` at first place is intended.
95 * `style` shares same memory with `num`, `ptr`, `color`.
96 * So we set the style value directly without using `prop.style.num`.
97 *
98 * E.g.
99 *
100 * static const lv_property_t obj_pos_x = {
101 * .id = LV_PROPERTY_STYLE_X,
102 * .num = 123,
103 * .selector = LV_STATE_PRESSED,
104 * }
105 *
106 * instead of:
107 * static const lv_property_t obj_pos_x = {
108 * .id = LV_PROPERTY_STYLE_X,
109 * .style.num = 123, // note this line.
110 * .selector = LV_STATE_PRESSED,
111 * }
112 */
113 lv_style_value_t style; /**< Make sure it's the first element in struct. */
114 uint32_t selector; /**< Style selector, lv_part_t | lv_state_t */
115 };
116 };
117} lv_property_t;
