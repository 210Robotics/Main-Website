262/**
263 * Calibrates the analog sensor on the specified port and returns the new
264 * calibration value.
265 *
266 * This method assumes that the true sensor value is not actively changing at
267 * this time and computes an average from approximately 500 samples, 1 ms apart,
268 * for a 0.5 s period of calibration. The average value thus calculated is
269 * returned and stored for later calls to the adi_analog_read_calibrated() and
270 * adi_analog_read_calibrated_HR() functions. These functions will return
271 * the difference between this value and the current sensor value when called.
272 *
273 * Do not use this function when the sensor value might be unstable
274 * (gyro rotation, accelerometer movement).
275 *
276 * \note The ADI currently returns data at 10ms intervals, in constrast to the
277 * calibrate function’s 1ms sample rate.
278 *
279 * This function uses the following values of errno when an error state is
280 * reached:
281 * ENXIO - The given value is not within the range of ADI Ports
282 *
283 * \param port
284 * The ADI port to calibrate (from 1-8, 'a'-'h', 'A'-'H')
285 *
286 * \return The average sensor value computed by this function
287 *
288 * \b Example
289 * \code
290 * #define ANALOG_SENSOR_PORT 1
291 *
292 * void initialize() {
293 * adi_analog_calibrate(ANALOG_SENSOR_PORT);
294 * printf("Calibrated Reading: %d\n", adi_analog_read_calibrated(ANALOG_SENSOR_PORT));
295 * // All readings from then on will be calibrated
296 * }
297 * \endcode
298 */
299int32_t adi_analog_calibrate(uint8_t port);

367/**
368 * Gets the 16 bit calibrated value of an analog input port.
369 *
370 * The adi_analog_calibrate() function must be run first. This is intended for
371 * integrated sensor values such as gyros and accelerometers to reduce drift due
372 * to round-off, and should not be used on a sensor such as a line tracker
373 * or potentiometer.
374 *
375 * The value returned actually has 16 bits of "precision", even though the ADC
376 * only reads 12 bits, so that error induced by the average value being between
377 * two values when integrated over time is trivial. Think of the value as the

379 *
380 * This function uses the following values of errno when an error state is
381 * reached:
382 * ENXIO - The given value is not within the range of ADI Ports
383 * EADDRINUSE - The port is not configured as an analog input
384 *
385 * \param port
386 * The ADI port (from 1-8, 'a'-'h', 'A'-'H') for which the value will be
387 * returned
388 *
389 * \return The difference of the sensor value from its calibrated default from
390 * -16384 to 16384
391 *
392 * \b Example
393 * \code
394 * #define ANALOG_SENSOR_PORT 1
395 *
396 * void opcontrol() {
397 * while (true) {
398 * adi_analog_calibrate(ANALOG_SENSOR_PORT);
399 * printf("Sensor Reading: %d\n", adi_analog_read_calibrated_HR(ANALOG_SENSOR_PORT));
