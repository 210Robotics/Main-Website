615 * \param line The line number on which to print
616 * \param x The (x,y) coordinates of the top left corner of the string
617 * \param y The (x,y) coordinates of the top left corner of the string
618 * \param fmt Format string
619 * \param ... Optional list of arguments for the format string
620 *
621 * \b Example
622 * \code
623 * void opcontrol() {
624 * int i = 0;
625 * pros::screen::set_pen(blue);
626 * while(1){
627 * // Will print seconds started since program started on line 3
628 * pros::screen::print(pros::TEXT_MEDIUM, 3, "Seconds Passed: %3d", i++);
629 * pros::delay(1000);
630 * }
631 * }
632 */

650 /**
651 * Gets the touch status of the last touch of the screen.
652 *
653 * \return The last_touch_e_t enum specifier that indicates the last touch status of the screen (E_TOUCH_EVENT_RELEASE, E_TOUCH_EVENT_PRESS, or E_TOUCH_EVENT_PRESS_AND_HOLD).
654 * This will be released by default if no action was taken.
655 * If an error occured, the screen_touch_status_s_t will have its
656 * last_touch_e_t enum specifier set to E_TOUCH_ERR, and other values set to -1.
657 *
658 * \b Example
659 * \code
660 * void opcontrol() {
661 * int i = 0;
662 * pros::screen_touch_status_s_t status;
663 * while(1){
664 * status = pros::touch_status();
665 *
666 * // Will print various information about the last touch
667 * pros::screen::print(TEXT_MEDIUM, 1, "Touch Status (Type): %d", status.touch_status);
668 * pros::screen::print(TEXT_MEDIUM, 2, "Last X: %d", status.x);
669 * pros::screen::print(TEXT_MEDIUM, 3, "Last Y: %d", status.y);
670 * pros::screen::print(TEXT_MEDIUM, 4, "Press Count: %d", status.press_count);
671 * pros::screen::print(TEXT_MEDIUM, 5, "Release Count: %d", status.release_count);
672 * pros::delay(20);
673 * }
674 * }
675 * \endcode
676 */
677 screen_touch_status_s_t touch_status();
