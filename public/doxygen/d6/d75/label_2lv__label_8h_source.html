47/** Long mode behaviors. Used in 'lv_label_ext_t'*/
48typedef enum {
49 LV_LABEL_LONG_WRAP, /**< Keep the object width, wrap lines longer than object width and expand the object height*/
50 LV_LABEL_LONG_DOT, /**< Keep the size and write dots at the end if the text is too long*/
51 LV_LABEL_LONG_SCROLL, /**< Keep the size and roll the text back and forth*/
52 LV_LABEL_LONG_SCROLL_CIRCULAR, /**< Keep the size and roll the text circularly*/
53 LV_LABEL_LONG_CLIP, /**< Keep the size and clip the text out of it*/
54} lv_label_long_mode_t;
