58/**
59 * Decode `full_area` pixels incrementally by calling in a loop. Set `decoded_area` values to `LV_COORD_MIN` on first call.
60 * Required only if the "open" function can't return with the whole decoded pixel array.
61 * @param decoder pointer to the decoder the function associated with
62 * @param dsc pointer to decoder descriptor
63 * @param full_area input parameter. the full area to decode after enough subsequent calls
64 * @param decoded_area input+output parameter. set the values to `LV_COORD_MIN` for the first call and to reset decoding.
65 * the decoded area is stored here after each call.
66 * @return LV_RESULT_OK: ok; LV_RESULT_INVALID: failed or there is nothing left to decode
67 */

69 lv_image_decoder_dsc_t * dsc,

83/**
84 * Get information about an image.
85 * Try the created image decoder one by one. Once one is able to get info that info will be used.
86 * @param src the image source. Can be
87 * 1) File name: E.g. "S:folder/img1.png" (The drivers needs to registered via `lv_fs_drv_register()`)
88 * 2) Variable: Pointer to an `lv_image_dsc_t` variable
89 * 3) Symbol: E.g. `LV_SYMBOL_OK`
90 * @param header the image info will be stored here
91 * @return LV_RESULT_OK: success; LV_RESULT_INVALID: wasn't able to get info about the image
92 */
93lv_result_t lv_image_decoder_get_info(const void * src, lv_image_header_t * header);
94
95/**
96 * Open an image.
97 * Try the created image decoders one by one. Once one is able to open the image that decoder is saved in `dsc`
98 * @param dsc describes a decoding session. Simply a pointer to an `lv_image_decoder_dsc_t` variable.
99 * @param src the image source. Can be
100 * 1) File name: E.g. "S:folder/img1.png" (The drivers needs to registered via `lv_fs_drv_register())`)
101 * 2) Variable: Pointer to an `lv_image_dsc_t` variable
102 * 3) Symbol: E.g. `LV_SYMBOL_OK`
103 * @param args args about how the image should be opened.
104 * @return LV_RESULT_OK: opened the image. `dsc->decoded` and `dsc->header` are set.
105 * LV_RESULT_INVALID: none of the registered image decoders were able to open the image.
106 */
107lv_result_t lv_image_decoder_open(lv_image_decoder_dsc_t * dsc, const void * src, const lv_image_decoder_args_t * args);
108
109/**
110 * Decode `full_area` pixels incrementally by calling in a loop. Set `decoded_area` to `LV_COORD_MIN` on first call.
111 * @param dsc image decoder descriptor
112 * @param full_area input parameter. the full area to decode after enough subsequent calls
113 * @param decoded_area input+output parameter. set the values to `LV_COORD_MIN` for the first call and to reset decoding.
114 * the decoded area is stored here after each call.
115 * @return LV_RESULT_OK: success; LV_RESULT_INVALID: an error occurred or there is nothing left to decode
