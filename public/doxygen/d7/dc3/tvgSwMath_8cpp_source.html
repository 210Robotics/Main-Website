50bool mathSmallCubic(const SwPoint* base, SwFixed& angleIn, SwFixed& angleMid, SwFixed& angleOut)
51{
52 auto d1 = base[2] - base[3];
53 auto d2 = base[1] - base[2];
54 auto d3 = base[0] - base[1];
55
56 if (d1.small()) {
57 if (d2.small()) {
58 if (d3.small()) {
59 angleIn = angleMid = angleOut = 0;
60 return true;
61 } else {
62 angleIn = angleMid = angleOut = mathAtan(d3);
63 }
64 } else {
65 if (d3.small()) {
66 angleIn = angleMid = angleOut = mathAtan(d2);
67 } else {
68 angleIn = angleMid = mathAtan(d2);
69 angleOut = mathAtan(d3);
70 }
71 }
72 } else {
73 if (d2.small()) {
74 if (d3.small()) {
75 angleIn = angleMid = angleOut = mathAtan(d1);
76 } else {
77 angleIn = mathAtan(d1);
78 angleOut = mathAtan(d3);
79 angleMid = mathMean(angleIn, angleOut);
80 }
81 } else {
82 if (d3.small()) {
83 angleIn = mathAtan(d1);
84 angleMid = angleOut = mathAtan(d2);
85 } else {
86 angleIn = mathAtan(d1);
87 angleMid = mathAtan(d2);
88 angleOut = mathAtan(d3);
89 }
90 }
91 }
92
93 auto theta1 = abs(mathDiff(angleIn, angleMid));
94 auto theta2 = abs(mathDiff(angleMid, angleOut));
95
96 if ((theta1 < (SW_ANGLE_PI / 8)) && (theta2 < (SW_ANGLE_PI / 8))) return true;
97 return false;
98}

101int64_t mathMultiply(int64_t a, int64_t b)
102{
103 int32_t s = 1;
104
105 //move sign
106 if (a < 0) {
107 a = -a;
108 s = -s;
109 }
110 if (b < 0) {
111 b = -b;
112 s = -s;
113 }
114 int64_t c = (a * b + 0x8000L) >> 16;
115 return (s > 0) ? c : -c;
116}

119int64_t mathDivide(int64_t a, int64_t b)
120{
121 int32_t s = 1;
122
123 //move sign
124 if (a < 0) {
125 a = -a;
126 s = -s;
127 }
128 if (b < 0) {
129 b = -b;
130 s = -s;
131 }
132 int64_t q = b > 0 ? ((a << 16) + (b >> 1)) / b : 0x7FFFFFFFL;
133 return (s < 0 ? -q : q);
134}

137int64_t mathMulDiv(int64_t a, int64_t b, int64_t c)
138{
139 int32_t s = 1;
140
141 //move sign
142 if (a < 0) {
143 a = -a;
144 s = -s;
145 }
146 if (b < 0) {
147 b = -b;
148 s = -s;
149 }
150 if (c < 0) {
151 c = -c;
152 s = -s;
153 }
154 int64_t d = c > 0 ? (a * b + (c >> 1)) / c : 0x7FFFFFFFL;
155
156 return (s > 0 ? d : -d);
157}

160void mathRotate(SwPoint& pt, SwFixed angle)
161{
162 if (angle == 0 || pt.zero()) return;
163
164 Point v = pt.toPoint();
165
166 auto radian = TO_RADIAN(angle);
167 auto cosv = cosf(radian);
168 auto sinv = sinf(radian);
169
170 pt.x = SwCoord(roundf((v.x * cosv - v.y * sinv) * 64.0f));
171 pt.y = SwCoord(roundf((v.x * sinv + v.y * cosv) * 64.0f));
172}

202SwFixed mathLength(const SwPoint& pt)
203{
204 if (pt.zero()) return 0;
205
206 //trivial case
207 if (pt.x == 0) return abs(pt.y);
208 if (pt.y == 0) return abs(pt.x);
209
210 auto v = pt.toPoint();
211 //return static_cast<SwFixed>(sqrtf(v.x * v.x + v.y * v.y) * 65536.0f);
212
213 /* approximate sqrt(x*x + y*y) using alpha max plus beta min algorithm.
214 With alpha = 1, beta = 3/8, giving results with the largest error less
215 than 7% compared to the exact value. */
216 if (v.x < 0) v.x = -v.x;
217 if (v.y < 0) v.y = -v.y;
218 return static_cast<SwFixed>((v.x > v.y) ? (v.x + v.y * 0.375f) : (v.y + v.x * 0.375f));
219}

222void mathSplitCubic(SwPoint* base)
223{
224 SwCoord a, b, c, d;
225
226 base[6].x = base[3].x;
227 c = base[1].x;
228 d = base[2].x;
229 base[1].x = a = (base[0].x + c) >> 1;
230 base[5].x = b = (base[3].x + d) >> 1;
231 c = (c + d) >> 1;
232 base[2].x = a = (a + c) >> 1;
233 base[4].x = b = (b + c) >> 1;
234 base[3].x = (a + b) >> 1;
235
236 base[6].y = base[3].y;
237 c = base[1].y;
238 d = base[2].y;
239 base[1].y = a = (base[0].y + c) >> 1;
240 base[5].y = b = (base[3].y + d) >> 1;
241 c = (c + d) >> 1;
242 base[2].y = a = (a + c) >> 1;
243 base[4].y = b = (b + c) >> 1;
244 base[3].y = (a + b) >> 1;
245}

248SwFixed mathDiff(SwFixed angle1, SwFixed angle2)
249{
250 auto delta = angle2 - angle1;
251
252 delta %= SW_ANGLE_2PI;
253 if (delta < 0) delta += SW_ANGLE_2PI;
254 if (delta > SW_ANGLE_PI) delta -= SW_ANGLE_2PI;
255
256 return delta;
257}

260SwPoint mathTransform(const Point* to, const Matrix* transform)
261{
262 if (!transform) return {TO_SWCOORD(to->x), TO_SWCOORD(to->y)};
263
264 auto tx = to->x * transform->e11 + to->y * transform->e12 + transform->e13;
265 auto ty = to->x * transform->e21 + to->y * transform->e22 + transform->e23;
266
267 return {TO_SWCOORD(tx), TO_SWCOORD(ty)};
268}

271bool mathClipBBox(const SwBBox& clipper, SwBBox& clipee)
272{
273 clipee.max.x = (clipee.max.x < clipper.max.x) ? clipee.max.x : clipper.max.x;
274 clipee.max.y = (clipee.max.y < clipper.max.y) ? clipee.max.y : clipper.max.y;
275 clipee.min.x = (clipee.min.x > clipper.min.x) ? clipee.min.x : clipper.min.x;
276 clipee.min.y = (clipee.min.y > clipper.min.y) ? clipee.min.y : clipper.min.y;
277
278 //Check valid region
279 if (clipee.max.x - clipee.min.x < 1 && clipee.max.y - clipee.min.y < 1) return false;
280
281 //Check boundary
282 if (clipee.min.x >= clipper.max.x || clipee.min.y >= clipper.max.y ||
283 clipee.max.x <= clipper.min.x || clipee.max.y <= clipper.min.y) return false;
284
285 return true;
286}

289bool mathUpdateOutlineBBox(const SwOutline* outline, const SwBBox& clipRegion, SwBBox& renderRegion, bool fastTrack)
290{
291 if (!outline) return false;
292
293 if (outline->pts.empty() || outline->cntrs.empty()) {
294 renderRegion.reset();
295 return false;
296 }
297
298 auto pt = outline->pts.begin();
299
300 auto xMin = pt->x;
301 auto xMax = pt->x;
302 auto yMin = pt->y;
303 auto yMax = pt->y;
304
305 for (++pt; pt < outline->pts.end(); ++pt) {
306 if (xMin > pt->x) xMin = pt->x;
307 if (xMax < pt->x) xMax = pt->x;
308 if (yMin > pt->y) yMin = pt->y;
309 if (yMax < pt->y) yMax = pt->y;
310 }
311 //Since no antialiasing is applied in the Fast Track case,
312 //the rasterization region has to be rearranged.
313 //https://github.com/Samsung/thorvg/issues/916
314 if (fastTrack) {
315 renderRegion.min.x = static_cast<SwCoord>(round(xMin / 64.0f));
316 renderRegion.max.x = static_cast<SwCoord>(round(xMax / 64.0f));
317 renderRegion.min.y = static_cast<SwCoord>(round(yMin / 64.0f));
318 renderRegion.max.y = static_cast<SwCoord>(round(yMax / 64.0f));
319 } else {
320 renderRegion.min.x = xMin >> 6;
321 renderRegion.max.x = (xMax + 63) >> 6;
322 renderRegion.min.y = yMin >> 6;
323 renderRegion.max.y = (yMax + 63) >> 6;
324 }
325 return mathClipBBox(clipRegion, renderRegion);
326}
