11namespace okapi {
12class EKFFilter : public Filter {
13 public:
14 /**
15 * One dimensional extended Kalman filter. The default arguments should work fine for most signal
16 * filtering. It won't hurt to graph your signal and the filtered result, and check if the filter
17 * is doing its job.
18 *
19 * Q is the covariance of the process noise and R is the
20 * covariance of the observation noise. The default values for Q and R should be a modest balance
21 * between trust in the sensor and FIR filtering.
22 *
23 * Think of R as how noisy your sensor is. Its value can be found mathematically by computing the
24 * standard deviation of your sensor reading vs. "truth" (of course, "truth" is still an estimate;
25 * try to calibrate your robot in a controlled setting where you can minimize the error in what
26 * "truth" is).
27 *
28 * Think of Q as how noisy your model is. It decides how much "smoothing" the filter does and how
29 * far it lags behind the true signal. This parameter is most often used as a "tuning" parameter
30 * to adjust the response of the filter.
31 *
32 * @param iQ process noise covariance
33 * @param iR measurement noise covariance
34 */
35 explicit EKFFilter(double iQ = 0.0001, double iR = ipow(0.2, 2));
36
37 /**
38 * Filters a value, like a sensor reading. Assumes the control input is zero.
39 *
40 * @param ireading new measurement
41 * @return filtered result
42 */
43 double filter(double ireading) override;
44
45 /**
46 * Filters a reading with a control input.
47 *
48 * @param ireading new measurement
49 * @param icontrol control input
50 * @return filtered result
51 */
52 virtual double filter(double ireading, double icontrol);
53
54 /**
55 * Returns the previous output from filter.
56 *
57 * @return the previous output from filter
58 */
59 double getOutput() const override;
60
61 protected:
62 const double Q, R;
63 double xHat = 0;
64 double xHatPrev = 0;
65 double xHatMinus = 0;
66 double P = 0;
67 double Pprev = 1;
68 double Pminus = 0;
69 double K = 0;
70};
71} // namespace okapi
