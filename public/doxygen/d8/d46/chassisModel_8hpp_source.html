16/**
17 * A version of the ReadOnlyChassisModel that also supports write methods, such as setting motor
18 * speed. Because this class can write to motors, there can only be one owner and as such copying
19 * is disabled.
20 */
21class ChassisModel : public ReadOnlyChassisModel {

34 /**
35 * Drive the robot in an arc (using open-loop control). Uses velocity mode.
36 * The algorithm is (approximately):
37 * leftPower = forwardSpeed + yaw
38 * rightPower = forwardSpeed - yaw
39 *
40 * @param iforwadSpeed speed in the forward direction
41 * @param iyaw speed around the vertical axis
42 */
43 virtual void driveVector(double iforwardSpeed, double iyaw) = 0;

49 * rightPower = forwardSpeed - yaw

51 * @param iforwadSpeed speed in the forward direction
52 * @param iyaw speed around the vertical axis
53 */
54 virtual void driveVectorVoltage(double iforwardSpeed, double iyaw) = 0;

59 * @param ispeed motor power

61 virtual void rotate(double ispeed) = 0;

63 /**
64 * Stop the robot (set all the motors to 0). Uses velocity mode.

66 virtual void stop() = 0;
67
68 /**

70 *
71 * @param ileftSpeed left side speed

75 virtual void tank(double ileftSpeed, double irightSpeed, double ithreshold = 0) = 0;

77 /**

79 *
80 * @param iforwardSpeed speed forward direction

83 */
84 virtual void arcade(double iforwardSpeed, double iyaw, double ithreshold = 0) = 0;

86 /**
87 * Drive the robot with a curvature drive layout. The robot drives in constant radius turns
88 * where you control the curvature (inverse of radius) you drive in. This is advantageous

91 *
92 * @param iforwardSpeed speed in the forward direction
93 * @param icurvature curvature (inverse of radius) to drive in
94 * @param ithreshold deadband on joystick values
95 */
96 virtual void curvature(double iforwardSpeed, double icurvature, double ithreshold = 0) = 0;
97
98 /**

103 virtual void left(double ispeed) = 0;
104
105 /**

108 * @param ipower motor power

110 virtual void right(double ispeed) = 0;
111
112 /**
113 * Reset the sensors to their zero point.
114 */
115 virtual void resetSensors() = 0;

117 /**

122 virtual void setBrakeMode(AbstractMotor::brakeMode mode) = 0;

124 /**

130
131 /**
132 * Set the gearset for each motor.
133 *
134 * @param gearset new motor gearset

136 virtual void setGearing(AbstractMotor::gearset gearset) = 0;

138 /**
139 * Sets a new maximum velocity in RPM. The usable maximum depends on the maximum velocity of the

142 * that velocity.
143 *
144 * @param imaxVelocity The new maximum velocity.

146 virtual void setMaxVelocity(double imaxVelocity) = 0;
147
148 /**
149 * @return The current maximum velocity.

153 /**

158 virtual void setMaxVoltage(double imaxVoltage) = 0;
159
160 /**
161 * @return The maximum voltage in mV `[0-12000]`.
162 */
163 virtual double getMaxVoltage() const = 0;
164};
