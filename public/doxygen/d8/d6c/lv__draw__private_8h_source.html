118 /**
119 * Called to signal the unit to complete all tasks in order to return their ready status.
120 * This callback can be implemented in case of asynchronous task processing.
121 * Below is an example to show the difference between synchronous and asynchronous:
122 *
123 * Synchronous:
124 * LVGL thread DRAW thread HW
125 *
126 * task1 --> submit --> Receive task1
127 * wait_for_finish()
128 * <-- task1->state = READY <-- Complete task1
129 * task2 --> submit --> Receive task2
130 * wait_for_finish()
131 * task2->state = READY <-- Complete task2
132 * task3 --> submit --> Receive task3
133 * wait_for_finish()
134 * <-- task3->state = READY <-- Complete task3
135 * task4 --> submit --> Receive task4
136 * wait_for_finish()
137 * <-- task4->state = READY <-- Complete task4
138 * NO MORE TASKS
139 *
140 *
141 * Asynchronous:
142 * LVGL thread DRAW thread HW
143 * is IDLE
144 * task1 --> queue task1
145 * submit --> Receive task1
146 * task2 --> queue task2 is BUSY (with task1)
147 * task3 --> queue task3 still BUSY (with task1)
148 * task4 --> queue task4 becomes IDLE
149 * <-- task1->state = READY <-- Complete task1
150 * submit --> Receive task2, task3, task4
151 * NO MORE TASKS
152 * wait_for_finish_cb() wait_for_finish()
153 * <-- Complete task2, task3, task4
154 * <-- task2->state = READY <--
155 * <-- task3->state = READY <--
156 * <-- task4->state = READY <--
157 *
158 * @param draw_unit
159 * @return
160 */
161 int32_t (*wait_for_finish_cb)(lv_draw_unit_t * draw_unit);
162
163 /**
