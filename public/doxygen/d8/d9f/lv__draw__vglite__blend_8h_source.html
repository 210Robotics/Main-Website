82typedef struct {

84 const lv_color_t * src; /**< Source buffer pointer (must be aligned on 32 bytes)*/
85 lv_area_t src_area; /**< Area to be copied from source*/
86 lv_coord_t src_width; /**< Source buffer width*/

88 int32_t src_stride; /**< Source buffer stride in bytes (must be aligned on 16 px)*/
89
90 const lv_color_t * dst; /**< Destination buffer pointer (must be aligned on 32 bytes)*/
91 lv_area_t dst_area; /**< Target area in destination buffer (must be the same as src_area)*/
92 lv_coord_t dst_width; /**< Destination buffer width*/
93 lv_coord_t dst_height; /**< Destination buffer height*/
94 int32_t dst_stride; /**< Destination buffer stride in bytes (must be aligned on 16 px)*/
95
96 lv_opa_t opa; /**< Opacity - alpha mix (0 = source not copied, 255 = 100% opaque)*/
97 uint32_t angle; /**< Rotation angle (1/10 of degree)*/
98 uint32_t zoom; /**< 256 = no zoom (1:1 scale ratio)*/
99 lv_point_t pivot; /**< The coordinates of rotation pivot in source image buffer*/
100} lv_gpu_nxp_vglite_blit_info_t;
