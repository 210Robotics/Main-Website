19class IMU : public ContinuousRotarySensor {
20 public:
21 /**
22 * An inertial sensor on the given port. The IMU returns an angle about the selected axis in
23 * degrees.
24 *
25 * ```cpp
26 * auto imuZ = IMU(1);
27 * auto imuX = IMU(1, IMUAxes::x);
28 * ```
29 *
30 * @param iport The port number in the range ``[1, 21]``.
31 * @param iaxis The axis of the inertial sensor to measure, default `IMUAxes::z`.
32 */
33 IMU(std::uint8_t iport, IMUAxes iaxis = IMUAxes::z);
34
35 /**
36 * Get the current rotation about the configured axis.
37 *
38 * @return The current sensor value or ``PROS_ERR``.
39 */
40 double get() const override;
41
42 /**
43 * Get the current sensor value remapped into the target range (``[-1800, 1800]`` by default).
44 *
45 * @param iupperBound The upper bound of the range.
46 * @param ilowerBound The lower bound of the range.
47 * @return The remapped sensor value.
48 */
49 double getRemapped(double iupperBound = 1800, double ilowerBound = -1800) const;
50
51 /**
52 * Get the current acceleration along the configured axis.
53 *
54 * @return The current sensor value or ``PROS_ERR``.
55 */
56 double getAcceleration() const;
57
58 /**
59 * Reset the rotation value to zero.
60 *
61 * @return ``1`` or ``PROS_ERR``.
62 */
63 std::int32_t reset() override;
64
65 /**
66 * Resets rotation value to desired value
67 * For example, ``reset(0)`` will reset the sensor to zero.
68 * But ``reset(90)`` will reset the sensor to 90 degrees.
69 *
70 * @param inewAngle desired reset value
71 * @return ``1`` or ``PROS_ERR``.
72 */
73 std::int32_t reset(double inewAngle);
74
75 /**
76 * Calibrate the IMU. Resets the rotation value to zero. Calibration is expected to take two
77 * seconds, but is bounded to five seconds.
78 *
79 * @return ``1`` or ``PROS_ERR``.
80 */
81 std::int32_t calibrate();
82
83 /**
84 * Get the sensor value for use in a control loop. This method might be automatically called in
85 * another thread by the controller.
86 *
87 * @return The current sensor value or ``PROS_ERR``.
88 */
89 double controllerGet() override;
90
91 /**
92 * @return Whether the IMU is calibrating.
93 */
94 bool isCalibrating() const;
95
96 protected:
97 std::uint8_t port;
98 IMUAxes axis;
99 double offset = 0;
100
101 /**
102 * Get the current rotation about the configured axis. The internal offset is not accounted for
103 * or modified. This just reads from the sensor.
104 *
105 * @return The current sensor value or ``PROS_ERR``.
106 */
107 double readAngle() const;
108};
109} // namespace okapi
