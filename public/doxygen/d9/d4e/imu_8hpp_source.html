91 /**
92 * Gets a IMU sensor that is plugged in to the brain
93 *
94 * \note The first time this function is called it returns the IMU sensor at the lowest port
95 * If this function is called multiple times, it will cycle through all the ports.
96 * For example, if you have 1 IMU sensor on the robot
97 * this function will always return a IMU sensor object for that port.
98 * If you have 2 IMU sensors, all the odd numered calls to this function will return objects
99 * for the lower port number,
100 * all the even number calls will return IMU objects for the higher port number
101 *
102 *
103 * This functions uses the following values of errno when an error state is
104 * reached:
105 * ENODEV - No IMU sensor is plugged into the brain
106 *
107 * \return A IMU object corresponding to a port that a IMU sensor is connected to the brain
108 * If no IMU sensor is plugged in, it returns a IMU sensor on port PROS_ERR_BYTE
109 *
110 */

112 /**
113 * Calibrate IMU
114 *
115 * Calibration takes approximately 2 seconds and blocks during this period if
116 * the blocking param is true, with a timeout for this operation being set a 3
117 * seconds as a safety margin. This function also blocks until the IMU
118 * status flag is set properly to E_IMU_STATUS_CALIBRATING, with a minimum
119 * blocking time of 5ms and a timeout of 1 second if it's never set.
120 *
121 * This function uses the following values of errno when an error state is
122 * reached:
123 * ENXIO - The given value is not within the range of V5 ports (1-21).
124 * ENODEV - The port cannot be configured as an Inertial Sensor
125 * EAGAIN - The sensor is already calibrating, or time out setting the status flag.
126 *
127 * \param blocking
128 * Whether this function blocks during calibration.
129 * \return 1 if the operation was successful or PROS_ERR if the operation
130 * failed, setting errno.
131 *
132 * \b Example
133 * \code
134 *
135 * #define IMU_PORT 1
136 *
137 * void opcontrol() {
138 * pros::Imu imu(IMU_PORT);
139 * imu.calibrate();
140 * // Block until calibration is complete
141 * imu.reset(true);
142 * }
143 * \endcode
144 */
145 virtual std::int32_t reset(bool blocking = false) const;
146 /**
147 * Set the Inertial Sensor's refresh interval in milliseconds.
148 *
149 * The rate may be specified in increments of 5ms, and will be rounded down to
150 * the nearest increment. The minimum allowable refresh rate is 5ms. The default
151 * rate is 10ms.
152 *
153 * As values are copied into the shared memory buffer only at 10ms intervals,
154 * setting this value to less than 10ms does not mean that you can poll the
155 * sensor's values any faster. However, it will guarantee that the data is as
156 * recent as possible.
157 *
158 * This function uses the following values of errno when an error state is
159 * reached:
160 * ENXIO - The given value is not within the range of V5 ports (1-21).
161 * ENODEV - The port cannot be configured as an Inertial Sensor
162 * EAGAIN - The sensor is still calibrating
163 *
164 * \param rate The data refresh interval in milliseconds
165 * \return 1 if the operation was successful or PROS_ERR if the operation
166 * failed, setting errno.
167 *
168 * \b Example
169 * \code
170 *
171 * #define IMU_PORT 1
172 *
173 * void opcontrol() {
174 * pros::Imu imu(IMU_PORT);
175 *
176 * while (true) {
177 * // Set the refresh rate to 5ms
178 * std::int32_t status = imu.set_data_rate(5);
179 * delay(20);
180 *
181 * // Check if the operation was successful
182 * if (status == PROS_ERR) {
183 * // Do something with the error
184 * }
185 *
186 * // Do something with the sensor data
187 * }
188 * }
189 * \endcode
190 */
191 virtual std::int32_t set_data_rate(std::uint32_t rate) const;
