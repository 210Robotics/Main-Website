20class OpticalSensor : public ControllerInput<double> {

22 /**
23 * An optical sensor on a V5 port.

25 * ```cpp
26 * auto osHue = OpticalSensor(1);

29 *
30 * @param iport The V5 port the device uses.
31 * @param ioutput Which sensor output to return from (@ref okapi::OpticalSensor::get).
