53/**
54 * \enum aivision_detected_type_e_t
55 * This enumeration defines what kind of object is stored inside the union in aivision_object_s
56 */
57typedef enum aivision_detected_type {
58 E_AIVISION_DETECTED_COLOR = (1 << 0),
59 E_AIVISION_DETECTED_CODE = (1 << 1),
60 E_AIVISION_DETECTED_OBJECT = (1 << 2),
61 E_AIVISION_DETECTED_TAG = (1 << 3)
62} aivision_detected_type_e_t;

64/**
65 * \enum aivision_mode_type_e_t
66 * This enumeration defines what kinds of objects the ai vision sensor will scan for:
67 * tags (april tags), colors (user defined colors), and objects (game elements), and all (all objects)
68 */
69
70typedef enum aivision_mode_type {
71 E_AIVISION_MODE_TAGS = (1 << 0),
72 E_AIVISION_MODE_COLORS = (1 << 1),
73 E_AIVISION_MODE_OBJECTS = (1 << 2),
74 E_AIVISION_MODE_COLOR_MERGE = (1 << 4),
75 E_AIVISION_MODE_ALL = (1 << 0) | (1 << 1) | (1 << 2),
76} aivision_mode_type_e_t;

78/**
79 * \struct aivision_color_s_t
80 * This structure contains the parameters used by the AI Vision Sensor to define a color. hue_range and saturation_range
81 * are ranges for hue and saturation that are acceptable.
82 * For example, if a large hue range is specified for a blue color, colors that are more magenta or teal may be detected
83 * as "blue".
84 */
85typedef struct aivision_color_s {
86 uint8_t id; /**< id of color descriptor, can range from 1-7 */
87 uint8_t red; /**< red value of color */
88 uint8_t green; /**< green value of color */
89 uint8_t blue; /**< blue value of color */
90 float hue_range; /**< range by which detected color's hue can vary from the base color, can range from 1-40 */
91 float saturation_range; /**< range by which detected color's saturation can vary from base color, can range from 0.1-1 */
92} aivision_color_s_t;
93
94/**
95 * \struct aivision_code_s_t
96 * This structure contains the parameters used by the AI Vision sensor to define a code.
97 * Codes are a combination of color descriptors, and tells the AI Vision sensor to merge objects
98 * close to each other that belong to the given color descriptors into a single object that matches
99 * the code descriptor.
100 * Codes must use at least 2, and no greater than 5, color descriptors.
101 */
102typedef struct aivision_code_s {
103 uint8_t id; /**< id of code descriptor, can range from 1-5 */
104 uint8_t length; /**< number of color descriptors used by this code. */
105 int16_t c1; /**< id of first color descriptor */
106 int16_t c2; /**< id of second color descriptor */
107 int16_t c3; /**< id of third color descriptor */
108 int16_t c4; /**< id of fourth color descriptor */
109 int16_t c5; /**< id of fifth color descriptor */
110} aivision_code_s_t;
111
112/**
113 * \enum aivision_tag_family_e_t
114 * This enumeration corresponds to a family of AprilTags.
115 * \see https://april.eecs.umich.edu/software/apriltag
116 */
117typedef enum aivision_tag_family_e {
118 TAG_CIRCLE_21H7 = 0,
119 TAG_16H5 = 1,
120 TAG_25H9 = 2,
121 TAG_61H11 = 3
122} aivision_tag_family_e_t;
123
124/**
125 * \struct aivision_object_color_s_t
126 * This structure contains a detected color.
127 */
128typedef struct __attribute__((packed)) aivision_object_color_s {
129 uint16_t xoffset; // left edge (from camera's view)
130 uint16_t yoffset; // top edge
131 uint16_t width;
132 uint16_t height;
133 uint16_t angle; // angle, in tenths of a degree
134} aivision_object_color_s_t;
135
136/**
137 * \struct aivision_object_tag_s_t
138 * This structure contains a detected tag.
139 */
140typedef struct __attribute__((packed)) aivision_object_tag_s {
141 int16_t x0;
142 int16_t y0;
143 int16_t x1;
144 int16_t y1;
145 int16_t x2;
146 int16_t y2;
147 int16_t x3;
148 int16_t y3;
149} aivision_object_tag_s_t;
150
151typedef struct __attribute__((packed)) aivision_object_element_s {
152 uint16_t xoffset; // left
153 uint16_t yoffset; // top
154 uint16_t width;
155 uint16_t height;
156 uint16_t score; // confidence that this struct is
157} aivision_object_element_s_t;
158/**
159 * \struct aivision_object_s_t
160 * This structure contains one of aivision_detected_type_e_t, stored in type
161 *
162 * If the object is a color, id stores the color's id
163 * If the object is an April Tag, id stores the tag's id
164 * If the object is an AI model element, id stores the element id as per
165 * https://api.vex.com/v5/home/cpp/AiVision/AiObjdesc.html
166 */
167typedef struct __attribute__((packed)) aivision_object_s {
168 uint8_t id; // object id
169 uint8_t type; // object type
170 union {
171 aivision_object_color_s_t color;
172 aivision_object_tag_s_t tag;
173 aivision_object_element_s_t element;
174 } object;
175} aivision_object_s_t;
