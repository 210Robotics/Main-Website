170/**
171 * @brief configure an axis to be tracked. Includes options for a delta threshold to prevent registering changes in axis state for small changes
172 * in axis value, and a deadzone to prevent registering changes in axis state when the axis is near the center position. Delta is also used for
173 * branch prediction and reducing the number of writes to the axis' state, so it can help with performance when used appropriately.
174 *
175 * @param axis the axis to configure
176 * @param delta the amount of change in axis value required to register a change in axis state
177 * @param deadzone the size of the deadzone for the axis
178 * @param errorBehavior the behavior to use when an error is encountered
179 */
180void GamepadController::configureAxis(Axis axis, std::int8_t delta, std::int8_t deadzone, axisErrorBehavior errorBehavior) {
181 if (checkIfAxisAlreadyConfigured(axis)) {
182 pros::lcd::print(0, "Failed to configure axis %s, axis already configured",
183 axisToString(axis).data());
184 return;
185 }
186 controllerAxes.at((std::int8_t)axis) = {axis, 0, 0, delta, deadzone, errorBehavior, true};
187}

189/**
190 * @brief refreshes all axis states, with respect to each axis' configured delta and deadzone values
191 *
192 */
193void GamepadController::updateAxisStates() {
194 for (GamepadAxis& a : controllerAxes) {
195 if (!a.configured) {
196 continue;
197 }
198 // Shortcut if simplified configuration is used
199 if (a.delta == 0 && a.deadzone == 0) {
200 a.value = getAxisState(a.axis, a.value, a.errorBehavior);
201 continue;
202 }
203
204 std::int8_t newValue = getAxisState(a.axis, a.value, a.errorBehavior);
205 // snaps new axis value to zero if within deadzone range
206 if (newValue < a.deadzone && newValue > -a.deadzone) {
207 newValue = 0;
208 }
209 int delta = std::abs(newValue - a.prevValue);
210 if (delta >= a.delta) {
211 // updates state, then previous state to match so new value is compared to the most recently registered value on the next update
212 a.value = newValue;
213 a.prevValue = newValue;
214 }
215 }
216}

222void GamepadController::updateBiStateButtonStates() {
223 // Unlike axis, I've curently not elected to use state-change detection for buttons,
224 // because we're only writing a boolean value
225 for (GamepadButton& b : biStateButtons) {
226 if (!b.configured) {
227 continue;
228 }
229 if (b.behavior == biStateButtonBehavior::HOLD) {
230 b.currentState = getButtonState(b.button, b.currentState, b.errorBehavior);
231 }
232 else if(b.behavior == biStateButtonBehavior::TOGGLE){
233 if(getButtonState(b.button, b.currentState, b.errorBehavior)){
234 b.currentState = !b.currentState;
235 }
236 }
237 else{
238 pros::lcd::print(0, "Error: unrecognized biStateButtonBehavior for button %s", buttonToString(b.button).data());
239 }

257/**
258 * @brief gets the current state of a button, returning true if the button is currently pressed, and false if it is not
259 *
260 * @param b
261 * @return true if the button was pressed
262 * @return false if button was not pressed, or if the button data was invalid
263 */
264bool GamepadController::getButtonState(Button b, bool previousState, buttonErrorBehavior errorBehavior) {
265 int result = 0;
266 switch (b) {
267 case Button::L1:
268 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_L1);
269 break;
270 case Button::L2:
271 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_L2);
272 break;
273 case Button::R1:
274 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_R1);
275 break;
276 case Button::R2:
277 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_R2);
278 break;
279 case Button::UP:
280 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_UP);
281 break;
282 case Button::DOWN:
283 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_DOWN);
284 break;
285 case Button::LEFT:
286 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_LEFT);
287 break;
288 case Button::RIGHT:
289 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_RIGHT);
290 break;
291 case Button::X:
292 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_X);
293 break;
294 case Button::B:
295 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_B);
296 break;
297 case Button::Y:
298 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_Y);
299 break;
300 case Button::A:
301 result = controller.get_digital(pros::E_CONTROLLER_DIGITAL_A);
302 break;
303 default:
304 pros::lcd::print(0, "Invalid button data");
305 result = 0;
306 }
307 switch (result) {
308 case 1:
309 return true;
310 case 0:
311 return false;
312 case EACCES:
313 pros::lcd::print(0, "Error: button not accessible");
314 if (errorBehavior == buttonErrorBehavior::KEEP_PREVIOUS_STATE) {
315 return previousState;
316 } else if (errorBehavior == buttonErrorBehavior::RELEASE) {
317 return false;
318 }
319 return false;
320 default:
321 pros::lcd::print(0, "Error occurred while getting button state");
322 return false;
323 }
324}

339/**
340 * @brief gets the current state of a tri-state button
341 *
342 * @param positive the positive button of the tri-state button pair
343 * @param negative the negative button of the tri-state button pair
344 * @return 1 if the positive button is pressed
345 * @return -1 if the negative button is pressed
346 * @return 0 if neither button is pressed, or if both buttons are pressed
347 */
348std::int8_t GamepadController::getTriStateButtonState(Button positive,
349 Button negative) {
350 TriStateButton& button = triStateButtons.at((std::int8_t)positive);
351 if (!button.configured) {
352 return 0;
353 }
354 bool positiveState = getButtonState(positive);
355 bool negativeState = getButtonState(negative);
356 // The first 2 conditions emulate how a d-pad works, where opposite directions cannot be pressed at the same time,
357 // which is the behavior needed here
358 if (positiveState && !negativeState) {
359 return 1;
360 } else if (!positiveState && negativeState) {
361 return -1;
362 } else {
363 return 0;
364 }
365}

381/**
382 * @brief gets the current state of an axis, private method that includes error handling for invalid axis states, and uses the axis' configured error behavior to determine how to handle errors.
383 *
384 * @param a the axis to get the state of, from [-127, 127]
385 * @param prevValue the previous value of the axis, used for error handling to prevent returning invalid values that could cause issues in the program
386 * @return the state of the axis(how far it is deflected)
387 */
388std::int8_t GamepadController::getAxisState(Axis a, int prevValue, axisErrorBehavior errorBehavior) {
389 std::int8_t result = 0;
390 switch (a) {
391 case Axis::LEFT_X:
392 result = controller.get_analog(pros::E_CONTROLLER_ANALOG_LEFT_X);
393 break;
394 case Axis::LEFT_Y:
395 result = controller.get_analog(pros::E_CONTROLLER_ANALOG_LEFT_Y);
396 break;
397 case Axis::RIGHT_X:
398 result = controller.get_analog(pros::E_CONTROLLER_ANALOG_RIGHT_X);
399 break;
400 case Axis::RIGHT_Y:
401 result = controller.get_analog(pros::E_CONTROLLER_ANALOG_RIGHT_Y);
402 break;
403 default:
404 pros::lcd::print(0, "Axis %d not mapped", axisToString(a).data());
405 result = 0;
406 }
407 if (result == EACCES) {
408 // Returns previous value so that if the axis is not accessible, the program can continue using the last known value of the axis instead of an invalid value
409 // This helps if an axis, for example, controls a motor, and the axis becomes inaccessible for some reason, so the motor will continue doing what it was last doing,
410 // so that the robot can still be controlled to some extent instead of becoming completely unresponsive, or killing power to a critical
411 // user action in a subsystem
412 // I personally don't think this will happen, but the get_analog function can return EACCES, so we must handle it
413 pros::lcd::print(0, "Error: axis not accessible");
414 return prevValue;
415 } else if (result < -127 || result > 127) {
416 pros::lcd::print(0, "Error: axis value out of range");
417 return 0;
418 } else {
419 return result;
420 }
421}
