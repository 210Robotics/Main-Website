62/**
63 * Open a file
64 * @param drv pointer to a driver where this function belongs
65 * @param path path to the file beginning with the driver letter (e.g. S:/folder/file.txt)
66 * @param mode read: FS_MODE_RD, write: FS_MODE_WR, both: FS_MODE_RD | FS_MODE_WR
67 * @return a file descriptor or NULL on error
68 */
69static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
70{
71 LV_UNUSED(drv);
72
73 const char * flags;
74 if(mode == LV_FS_MODE_WR)
75 flags = FILE_WRITE;
76 else if(mode == LV_FS_MODE_RD)
77 flags = FILE_READ;
78 else if(mode == (LV_FS_MODE_WR | LV_FS_MODE_RD))
79 flags = FILE_WRITE;
80
81 File file = SD.open(path, flags);
82 if(!file) {
83 return NULL;
84 }
85
86 SdFile * lf = new SdFile{file};
87
88 return (void *)lf;
89}

143/**
144 * Set the read write pointer. Also expand the file size if necessary.
145 * @param drv pointer to a driver where this function belongs
146 * @param file_p pointer to a file_t variable. (opened with fs_open )
147 * @param pos the new position of read write pointer
148 * @param whence tells from where to interpret the `pos`. See @lv_fs_whence_t
149 * @return LV_FS_RES_OK: no error or any error from @lv_fs_res_t enum
150 */
151static lv_fs_res_t fs_seek(lv_fs_drv_t * drv, void * file_p, uint32_t pos, lv_fs_whence_t whence)
152{
153 LV_UNUSED(drv);
154 SeekMode mode;
155 if(whence == LV_FS_SEEK_SET)
156 mode = SeekSet;
157 else if(whence == LV_FS_SEEK_CUR)
158 mode = SeekCur;
159 else if(whence == LV_FS_SEEK_END)
160 mode = SeekEnd;
161
162 SdFile * lf = (SdFile *)file_p;
163
164 int rc = lf->file.seek(pos, mode);
165
166 return rc < 0 ? LV_FS_RES_UNKNOWN : LV_FS_RES_OK;
167}
