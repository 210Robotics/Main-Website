108/**
109 * Set the GPS's offset relative to the center of turning in meters,
110 * as well as its initial position.
111 *
112 * This function uses the following values of errno when an error state is
113 * reached:
114 * ENXIO - The given value is not within the range of V5 ports (1-21).
115 * ENODEV - The port cannot be configured as a GPS
116 * EAGAIN - The sensor is still calibrating
117 *
118 * \param port
119 * The V5 GPS port number from 1-21
120 * \param xOffset
121 * Cartesian 4-Quadrant X offset from center of turning (meters)
122 * \param yOffset
123 * Cartesian 4-Quadrant Y offset from center of turning (meters)
124 * \param xInitial
125 * Initial 4-Quadrant X Position, with (0,0) being at the center of the field (meters)
126 * \param yInitial
127 * Initial 4-Quadrant Y Position, with (0,0) being at the center of the field (meters)
128 * \param headingInitial
129 * Heading with 0 being north on the field, in degrees [0,360) going clockwise
130 * \return 1 if the operation was successful or PROS_ERR if the operation
131 * failed, setting errno.
132 *
133 * \b Example
134 * \code
135 * #define GPS_PORT 1
136 * #define X_OFFSET .225
137 * #define Y_OFFSET .223
138 * #define X_INITIAL 1.54
139 * #define Y_INITIAL 1.14
140 * #define HEADING_INITIAL 90
141 *
142 * void initialize() {
143 * gps_initialize_full(GPS_PORT, X_OFFSET, Y_OFFSET, X_INITIAL, Y_INITIAL, HEADING_INITIAL);
144 * }
145 * \endcode
146 */
147int32_t gps_initialize_full(uint8_t port, double xInitial, double yInitial, double headingInitial, double xOffset,
148 double yOffset);
