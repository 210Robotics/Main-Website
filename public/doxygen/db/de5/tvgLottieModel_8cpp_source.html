41void LottieSlot::reset()
42{
43 if (!overridden) return;
44
45 for (auto pair = pairs.begin(); pair < pairs.end(); ++pair) {
46 switch (type) {
47 case LottieProperty::Type::ColorStop: {
48 static_cast<LottieGradient*>(pair->obj)->colorStops.release();
49 static_cast<LottieGradient*>(pair->obj)->colorStops = *static_cast<LottieColorStop*>(pair->prop);
50 static_cast<LottieColorStop*>(pair->prop)->frames = nullptr;
51 break;
52 }
53 case LottieProperty::Type::Color: {
54 static_cast<LottieSolid*>(pair->obj)->color.release();
55 static_cast<LottieSolid*>(pair->obj)->color = *static_cast<LottieColor*>(pair->prop);
56 static_cast<LottieColor*>(pair->prop)->frames = nullptr;
57 break;
58 }
59 case LottieProperty::Type::TextDoc: {
60 static_cast<LottieText*>(pair->obj)->doc.release();
61 static_cast<LottieText*>(pair->obj)->doc = *static_cast<LottieTextDoc*>(pair->prop);
62 static_cast<LottieTextDoc*>(pair->prop)->frames = nullptr;
63 break;
64 }
65 default: break;
66 }
67 delete(pair->prop);
68 pair->prop = nullptr;
69 }
70 overridden = false;
71}

74void LottieSlot::assign(LottieObject* target)
75{
76 //apply slot object to all targets
77 for (auto pair = pairs.begin(); pair < pairs.end(); ++pair) {
78 //backup the original properties before overwriting
79 switch (type) {
80 case LottieProperty::Type::ColorStop: {
81 if (!overridden) {
82 pair->prop = new LottieColorStop;
83 *static_cast<LottieColorStop*>(pair->prop) = static_cast<LottieGradient*>(pair->obj)->colorStops;
84 }
85
86 pair->obj->override(&static_cast<LottieGradient*>(target)->colorStops);
87 break;
88 }
89 case LottieProperty::Type::Color: {
90 if (!overridden) {
91 pair->prop = new LottieColor;
92 *static_cast<LottieColor*>(pair->prop) = static_cast<LottieSolid*>(pair->obj)->color;
93 }
94
95 pair->obj->override(&static_cast<LottieSolid*>(target)->color);
96 break;
97 }
98 case LottieProperty::Type::TextDoc: {
99 if (!overridden) {
100 pair->prop = new LottieTextDoc;
101 *static_cast<LottieTextDoc*>(pair->prop) = static_cast<LottieText*>(pair->obj)->doc;
102 }
103
104 pair->obj->override(&static_cast<LottieText*>(target)->doc);
105 break;
106 }
107 default: break;
108 }
109 }
110 overridden = true;
111}

125void LottieTrimpath::segment(float frameNo, float& start, float& end, LottieExpressions* exps)
126{
127 auto s = this->start(frameNo, exps) * 0.01f;
128 auto e = this->end(frameNo, exps) * 0.01f;
129 auto o = fmodf(this->offset(frameNo, exps), 360.0f) / 360.0f; //0 ~ 1
130
131 auto diff = fabs(s - e);
132 if (mathZero(diff)) {
133 start = 0.0f;
134 end = 0.0f;
135 return;
136 }
137 if (mathEqual(diff, 1.0f) || mathEqual(diff, 2.0f)) {
138 start = 0.0f;
139 end = 1.0f;
140 return;
141 }
142
143 s += o;
144 e += o;
145
146 auto loop = true;
147
148 //no loop
149 if (s > 1.0f && e > 1.0f) loop = false;
150 if (s < 0.0f && e < 0.0f) loop = false;
151 if (s >= 0.0f && s <= 1.0f && e >= 0.0f && e <= 1.0f) loop = false;
152
153 if (s > 1.0f) s -= 1.0f;
154 if (s < 0.0f) s += 1.0f;
155 if (e > 1.0f) e -= 1.0f;
156 if (e < 0.0f) e += 1.0f;
157
158 if (loop) {
159 start = s > e ? s : e;
160 end = s < e ? s : e;
161 } else {
162 start = s < e ? s : e;
163 end = s > e ? s : e;
164 }
165}

168uint32_t LottieGradient::populate(ColorStop& color)
169{
170 colorStops.populated = true;
171 if (!color.input) return 0;
172
173 uint32_t alphaCnt = (color.input->count - (colorStops.count * 4)) / 2;
174 Array<Fill::ColorStop> output(colorStops.count + alphaCnt);
175 uint32_t cidx = 0; //color count
176 uint32_t clast = colorStops.count * 4;
177 if (clast > color.input->count) clast = color.input->count;
178 uint32_t aidx = clast; //alpha count
179 Fill::ColorStop cs;
180
181 //merge color stops.
182 for (uint32_t i = 0; i < color.input->count; ++i) {
183 if (cidx == clast || aidx == color.input->count) break;
184 if ((*color.input)[cidx] == (*color.input)[aidx]) {
185 cs.offset = (*color.input)[cidx];
186 cs.r = lroundf((*color.input)[cidx + 1] * 255.0f);
187 cs.g = lroundf((*color.input)[cidx + 2] * 255.0f);
188 cs.b = lroundf((*color.input)[cidx + 3] * 255.0f);
189 cs.a = lroundf((*color.input)[aidx + 1] * 255.0f);
190 cidx += 4;
191 aidx += 2;
192 } else if ((*color.input)[cidx] < (*color.input)[aidx]) {
193 cs.offset = (*color.input)[cidx];
194 cs.r = lroundf((*color.input)[cidx + 1] * 255.0f);
195 cs.g = lroundf((*color.input)[cidx + 2] * 255.0f);
196 cs.b = lroundf((*color.input)[cidx + 3] * 255.0f);
197 //generate alpha value
198 if (output.count > 0) {
199 auto p = ((*color.input)[cidx] - output.last().offset) / ((*color.input)[aidx] - output.last().offset);
200 cs.a = mathLerp<uint8_t>(output.last().a, lroundf((*color.input)[aidx + 1] * 255.0f), p);
201 } else cs.a = 255;
202 cidx += 4;
203 } else {
204 cs.offset = (*color.input)[aidx];
205 cs.a = lroundf((*color.input)[aidx + 1] * 255.0f);
206 //generate color value
207 if (output.count > 0) {
208 auto p = ((*color.input)[aidx] - output.last().offset) / ((*color.input)[cidx] - output.last().offset);
209 cs.r = mathLerp<uint8_t>(output.last().r, lroundf((*color.input)[cidx + 1] * 255.0f), p);
210 cs.g = mathLerp<uint8_t>(output.last().g, lroundf((*color.input)[cidx + 2] * 255.0f), p);
211 cs.b = mathLerp<uint8_t>(output.last().b, lroundf((*color.input)[cidx + 3] * 255.0f), p);
212 } else cs.r = cs.g = cs.b = 255;
213 aidx += 2;
214 }
215 output.push(cs);
216 }
217
218 //color remains
219 while (cidx + 3 < clast) {
220 cs.offset = (*color.input)[cidx];
221 cs.r = lroundf((*color.input)[cidx + 1] * 255.0f);
222 cs.g = lroundf((*color.input)[cidx + 2] * 255.0f);
223 cs.b = lroundf((*color.input)[cidx + 3] * 255.0f);
224 cs.a = (output.count > 0) ? output.last().a : 255;
225 output.push(cs);
226 cidx += 4;
227 }
228
229 //alpha remains
230 while (aidx < color.input->count) {
231 cs.offset = (*color.input)[aidx];
232 cs.a = lroundf((*color.input)[aidx + 1] * 255.0f);
233 if (output.count > 0) {
234 cs.r = output.last().r;
235 cs.g = output.last().g;
236 cs.b = output.last().b;
237 } else cs.r = cs.g = cs.b = 255;
238 output.push(cs);
239 aidx += 2;
240 }
241
242 color.data = output.data;
243 output.data = nullptr;
244
245 color.input->reset();
246 delete(color.input);
247
248 return output.count;
249}

252Fill* LottieGradient::fill(float frameNo, LottieExpressions* exps)
253{
254 Fill* fill = nullptr;
255 auto s = start(frameNo, exps);
256 auto e = end(frameNo, exps);
257
258 //Linear Gradient
259 if (id == 1) {
260 fill = LinearGradient::gen().release();
261 static_cast<LinearGradient*>(fill)->linear(s.x, s.y, e.x, e.y);
262 }
263 //Radial Gradient
264 if (id == 2) {
265 fill = RadialGradient::gen().release();
266
267 auto w = fabsf(e.x - s.x);
268 auto h = fabsf(e.y - s.y);
269 auto r = (w > h) ? (w + 0.375f * h) : (h + 0.375f * w);
270 auto progress = this->height(frameNo, exps) * 0.01f;
271
272 if (mathZero(progress)) {
273 P(static_cast<RadialGradient*>(fill))->radial(s.x, s.y, r, s.x, s.y, 0.0f);
274 } else {
275 if (mathEqual(progress, 1.0f)) progress = 0.99f;
276 auto startAngle = mathRad2Deg(atan2(e.y - s.y, e.x - s.x));
277 auto angle = mathDeg2Rad((startAngle + this->angle(frameNo, exps)));
278 auto fx = s.x + cos(angle) * progress * r;
279 auto fy = s.y + sin(angle) * progress * r;
280 // Lottie doesn't have any focal radius concept
281 P(static_cast<RadialGradient*>(fill))->radial(s.x, s.y, r, fx, fy, 0.0f);
282 }
283 }
284
285 if (!fill) return nullptr;
286
287 colorStops(frameNo, fill, exps);
288
289 return fill;
290}

293void LottieGroup::prepare(LottieObject::Type type)
294{
295 LottieObject::type = type;
296
297 if (children.count == 0) return;
298
299 size_t strokeCnt = 0;
300 size_t fillCnt = 0;
301
302 for (auto c = children.end() - 1; c >= children.begin(); --c) {
303 auto child = static_cast<LottieObject*>(*c);
304
305 if (child->type == LottieObject::Type::Trimpath) trimpath = true;
306
307 /* Figure out if this group is a simple path drawing.
308 In that case, the rendering context can be sharable with the parent's. */
309 if (allowMerge && (child->type == LottieObject::Group || !child->mergeable())) allowMerge = false;
310
311 if (reqFragment) continue;
312
313 /* Figure out if the rendering context should be fragmented.
314 Multiple stroking or grouping with a stroking would occur this.
315 This fragment resolves the overlapped stroke outlines. */
316 if (child->type == LottieObject::Group && !child->mergeable()) {
317 if (strokeCnt > 0 || fillCnt > 0) reqFragment = true;
318 } else if (child->type == LottieObject::SolidStroke || child->type == LottieObject::GradientStroke) {
319 if (strokeCnt > 0) reqFragment = true;
320 else ++strokeCnt;
321 } else if (child->type == LottieObject::SolidFill || child->type == LottieObject::GradientFill) {
322 if (fillCnt > 0) reqFragment = true;
323 else ++fillCnt;
324 }
325 }
326
327 //Reverse the drawing order if this group has a trimpath.
328 if (!trimpath) return;
329
330 for (uint32_t i = 0; i < children.count - 1; ) {
331 auto child2 = children[i + 1];
332 if (!child2->mergeable() || child2->type == LottieObject::Transform) {
333 i += 2;
334 continue;
335 }
336 auto child = children[i];
337 if (!child->mergeable() || child->type == LottieObject::Transform) {
338 i++;
339 continue;
340 }
341 children[i] = child2;
342 children[i + 1] = child;
343 i++;
344 }
345}

364void LottieLayer::prepare()
365{
366 /* if layer is hidden, only useful data is its transform matrix.
367 so force it to be a Null Layer and release all resource. */
368 if (hidden) {
369 type = LottieLayer::Null;
370 for (auto p = children.begin(); p < children.end(); ++p) delete(*p);
371 children.reset();
372 return;
373 }
374 LottieGroup::prepare(LottieObject::Layer);

378float LottieLayer::remap(float frameNo, LottieExpressions* exp)
379{
380 if (timeRemap.frames || timeRemap.value) {
381 frameNo = comp->frameAtTime(timeRemap(frameNo, exp));
