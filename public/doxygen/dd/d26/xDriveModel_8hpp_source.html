43 /**
44 * Drive the robot in an arc (using open-loop control). Uses velocity mode.
45 * The algorithm is (approximately):
46 * leftPower = forwardSpeed + yaw
47 * rightPower = forwardSpeed - yaw
48 *
49 * @param iforwardSpeed speed in the forward direction
50 * @param iyaw speed around the vertical axis
51 */
52 void driveVector(double iforwardSpeed, double iyaw) override;

54 /**

56 * The algorithm is (approximately):
57 * leftPower = forwardSpeed + yaw
58 * rightPower = forwardSpeed - yaw
59 *
60 * @param iforwadSpeed speed in the forward direction
61 * @param iyaw speed around the vertical axis
62 */
63 void driveVectorVoltage(double iforwardSpeed, double iyaw) override;

80 /**
81 * Strafe the robot in an arc (using open-loop control) where positive istrafeSpeed is
82 * to the right and negative istrafeSpeed is to the left. Uses velocity mode.
83 * The algorithm is (approximately):
84 * topLeftPower = -1 * istrafeSpeed + yaw
85 * topRightPower = istrafeSpeed - yaw
86 * bottomRightPower = -1 * istrafeSpeed - yaw
87 * bottomLeftPower = istrafeSpeed + yaw
88 *
89 * @param istrafeSpeed speed to the right
90 * @param iyaw speed around the vertical axis
91 */
92 void strafeVector(double istrafeSpeed, double iyaw);
93
94 /**

97 void stop() override;

99 /**
100 * Drive the robot with a tank drive layout. Uses voltage mode.

102 * @param ileftSpeed left side speed

106 void tank(double ileftSpeed, double irightSpeed, double ithreshold = 0) override;

112 * @param iyaw speed around the vertical axis
113 * @param ithreshold deadband on joystick values

117 /**
118 * Drive the robot with a curvature drive layout. The robot drives in constant radius turns
119 * where you control the curvature (inverse of radius) you drive in. This is advantageous
120 * because the forward speed will not affect the rate of turning. The algorithm switches to
121 * arcade if the forward speed is 0. Uses voltage mode.
122 *
123 * @param iforwardSpeed speed forward direction
124 * @param icurvature curvature (inverse of radius) to drive in
125 * @param ithreshold deadband on joystick values
126 */
127 void curvature(double iforwardSpeed, double icurvature, double ithreshold = 0) override;
128
129 /**
130 * Drive the robot with an arcade drive layout. Uses voltage mode.
131 *
132 * @param irightSpeed speed to the right
133 * @param iforwardSpeed speed in the forward direction
134 * @param iyaw speed around the vertical axis
135 * @param ithreshold deadband on joystick values
136 */
137 virtual void
138 xArcade(double irightSpeed, double iforwardSpeed, double iyaw, double ithreshold = 0);
139
140 /**
141 * Drive the robot with a field-oriented arcade drive layout. Uses voltage mode.
142 * For example:
143 * Both `fieldOrientedXArcade(1, 0, 0, 0_deg)` and `fieldOrientedXArcade(1, 0, 0, 90_deg)`
144 * will drive the chassis in the forward/north direction. In other words, no matter
145 * the robot's heading, the robot will move forward/north when you tell it
146 * to move forward/north and will move right/east when you tell it to move right/east.
147 *
148 *
149 * @param ixSpeed forward speed -- (`+1`) forward, (`-1`) backward
150 * @param iySpeed sideways speed -- (`+1`) right, (`-1`) left
151 * @param iyaw turn speed -- (`+1`) clockwise, (`-1`) counter-clockwise
152 * @param iangle current chassis angle (`0_deg` = no correction, winds clockwise)
153 * @param ithreshold deadband on joystick values
154 */
155 virtual void fieldOrientedXArcade(double ixSpeed,
156 double iySpeed,
157 double iyaw,
158 QAngle iangle,
159 double ithreshold = 0);
