45/**
46 * Sets the voltage for the motor from -127 to 127.
47 *
48 * This is designed to map easily to the input from the controller's analog
49 * stick for simple opcontrol use. The actual behavior of the motor is analogous
50 * to use of motor_move_voltage().
51 *
52 * \note This function will not respect brake modes, and simply sets the voltage to the desired value.
53 *
54 * \note A negative port will negate the input voltage
55 *
56 * This function uses the following values of errno when an error state is
57 * reached:
58 * ENXIO - The given value is not within the range of V5 ports |1-21|.
59 * ENODEV - The port cannot be configured as a motor
60 *
61 * \param port
62 * The V5 port number from 1 to 21, or from -21 to -1 for reversed motors
63 * \param voltage
64 * The new motor voltage from -127 to 127
65 *
66 * \return 1 if the operation was successful or PROS_ERR if the operation
67 * failed, setting errno.
68 *
69 * \b Example
70 * \code
71 * void opcontrol() {
72 * while (true) {
73 * motor_move(1, controller_get_analog(E_CONTROLLER_MASTER, E_CONTROLLER_ANALOG_LEFT_Y));
74 * delay(2);
75 * }
76 * }
77 * \endcode
78 */

111/**
112 * Sets the target absolute position for the motor to move to.
113 *
114 * This movement is relative to the position of the motor when initialized or
115 * the position when it was most recently reset with motor_set_zero_position().
116 *
117 * \note This function simply sets the target for the motor, it does not block program
118 * execution until the movement finishes. The example code shows how to block until a movement is finished.
119 *
120 * \note A negative port number will negate the target position
121 *
122 * This function uses the following values of errno when an error state is
123 * reached:
124 * ENXIO - The given value is not within the range of V5 ports |1-21|.
125 * ENODEV - The port cannot be configured as a motor
126 *
127 * \param port
128 * The V5 port number from 1 to 21, or from -21 to -1 for reversed motors
129 * \param position
130 * The absolute position to move to in the motor's encoder units
131 * \param velocity
132 * The maximum allowable velocity for the movement in RPM
133 *
134 * \return 1 if the operation was successful or PROS_ERR if the operation
135 * failed, setting errno.
136 *
137 * \b Example
138 * \code
139 * void autonomous() {
140 * motor_move_absolute(1, 100, 100); // Moves 100 units forward
141 * while (!((motor_get_position(1) < 105) && (motor_get_position(1) > 95))) {
142 * // Continue running this loop as long as the motor is not within +-5 units of its goal
143 * delay(2);
144 * }
145 * motor_move_absolute(1, 100, 100); // This will not cause a movement
146 * while(!((motor_get_position(1) < 105) && (motor_get_position(1) > 95))) {
147 * delay(2);
148 * }
149 *
150 * motor_tare_position(1);
151 * motor_move_absolute(1, 100, 100); // Moves 100 units forward
