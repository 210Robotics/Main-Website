39struct Text::Impl
40{
41 FontLoader* loader = nullptr;
42 Shape* paint = nullptr;
43 char* utf8 = nullptr;
44 float fontSize;
45 bool italic = false;
46 bool changed = false;
47
48 ~Impl()
49 {
50 free(utf8);
51 LoaderMgr::retrieve(loader);
52 delete(paint);
53 }
54
55 Result fill(uint8_t r, uint8_t g, uint8_t b)
56 {
57 return paint->fill(r, g, b);
58 }
59
60 Result fill(Fill* f)
61 {
62 return paint->fill(cast<Fill>(f));
63 }
64
65 Result text(const char* utf8)
66 {
67 free(this->utf8);
68 if (utf8) this->utf8 = strdup(utf8);
69 else this->utf8 = nullptr;
70 changed = true;
71
72 return Result::Success;
73 }
74
75 Result font(const char* name, float size, const char* style)
76 {
77 auto loader = LoaderMgr::loader(name);
78 if (!loader) return Result::InsufficientCondition;
79
80 //Same resource has been loaded.
81 if (this->loader == loader) {
82 this->loader->sharing--; //make it sure the reference counting.
83 return Result::Success;
84 } else if (this->loader) {
85 LoaderMgr::retrieve(this->loader);
86 }
87 this->loader = static_cast<FontLoader*>(loader);
88
89 if (!paint) paint = Shape::gen().release();
90
91 fontSize = size;
92 if (style && strstr(style, "italic")) italic = true;
93 changed = true;
94 return Result::Success;
95 }
96
97 RenderRegion bounds(RenderMethod* renderer)
98 {
99 if (paint) return P(paint)->bounds(renderer);
100 else return {0, 0, 0, 0};
101 }
102
103 bool render(RenderMethod* renderer)
104 {
105 if (paint) return PP(paint)->render(renderer);
106 return false;
107 }
108
109 bool load()
110 {
111 if (!loader) return false;
112
113 //reload
114 if (changed) {
115 loader->request(paint, utf8, italic);
116 loader->read();
117 changed = false;
118 }
119 if (paint) {
120 loader->resize(paint, fontSize, fontSize);
121 return true;
122 }
123 return false;
124 }
125
126 RenderData update(RenderMethod* renderer, const RenderTransform* transform, Array<RenderData>& clips, uint8_t opacity, RenderUpdateFlag pFlag, bool clipper)
127 {
128 if (!load()) return nullptr;
129
130 //transform the gradient coordinates based on the final scaled font.
131 if (P(paint)->flag & RenderUpdateFlag::Gradient) {
132 auto fill = P(paint)->rs.fill;
133 auto scale = 1.0f / loader->scale;
134 if (fill->identifier() == TVG_CLASS_ID_LINEAR) {
135 P(static_cast<LinearGradient*>(fill))->x1 *= scale;
136 P(static_cast<LinearGradient*>(fill))->y1 *= scale;
137 P(static_cast<LinearGradient*>(fill))->x2 *= scale;
138 P(static_cast<LinearGradient*>(fill))->y2 *= scale;
139 } else {
140 P(static_cast<RadialGradient*>(fill))->cx *= scale;
141 P(static_cast<RadialGradient*>(fill))->cy *= scale;
142 P(static_cast<RadialGradient*>(fill))->r *= scale;
143 P(static_cast<RadialGradient*>(fill))->fx *= scale;
144 P(static_cast<RadialGradient*>(fill))->fy *= scale;
145 P(static_cast<RadialGradient*>(fill))->fr *= scale;
146 }
147 }
148 return PP(paint)->update(renderer, transform, clips, opacity, pFlag, clipper);
149 }
150
151 bool bounds(float* x, float* y, float* w, float* h, TVG_UNUSED bool stroking)
152 {
153 if (!load() || !paint) return false;
154 paint->bounds(x, y, w, h, true);
155 return true;
156 }
157
158 Paint* duplicate()
159 {
160 load();
161
162 auto ret = Text::gen().release();
163 auto dup = ret->pImpl;
164 if (paint) dup->paint = static_cast<Shape*>(paint->duplicate());
165
166 if (loader) {
167 dup->loader = loader;
168 ++dup->loader->sharing;
169 }
170
171 dup->utf8 = strdup(utf8);
172 dup->italic = italic;
173 dup->fontSize = fontSize;
174
175 return ret;
176 }
177
178 Iterator* iterator()
179 {
180 return nullptr;
181 }
182};
