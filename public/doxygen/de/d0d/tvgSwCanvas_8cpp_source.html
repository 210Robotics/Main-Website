66Result SwCanvas::mempool(MempoolPolicy policy) noexcept
67{
68#ifdef THORVG_SW_RASTER_SUPPORT
69 //We know renderer type, avoid dynamic_cast for performance.
70 auto renderer = static_cast<SwRenderer*>(Canvas::pImpl->renderer);
71 if (!renderer) return Result::MemoryCorruption;
72
73 //It can't change the policy during the running.
74 if (!Canvas::pImpl->paints.empty()) return Result::InsufficientCondition;
75
76 if (policy == MempoolPolicy::Individual) renderer->mempool(false);
77 else renderer->mempool(true);
78
79 return Result::Success;
80#endif
81 return Result::NonSupport;
82}

85Result SwCanvas::target(uint32_t* buffer, uint32_t stride, uint32_t w, uint32_t h, Colorspace cs) noexcept
86{
87#ifdef THORVG_SW_RASTER_SUPPORT
88 //We know renderer type, avoid dynamic_cast for performance.
89 auto renderer = static_cast<SwRenderer*>(Canvas::pImpl->renderer);
90 if (!renderer) return Result::MemoryCorruption;
91
92 if (!renderer->target(buffer, stride, w, h, static_cast<ColorSpace>(cs))) return Result::InvalidArguments;
93 Canvas::pImpl->vport = {0, 0, (int32_t)w, (int32_t)h};
94 renderer->viewport(Canvas::pImpl->vport);
95
96 //Paints must be updated again with this new target.
97 Canvas::pImpl->needRefresh();
98
99 //FIXME: The value must be associated with an individual canvas instance.
100 ImageLoader::cs = static_cast<ColorSpace>(cs);
101
102 return Result::Success;
103#endif
104 return Result::NonSupport;
105}
