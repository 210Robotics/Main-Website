184/**
185 * Calibrates the analog sensor on the specified port and returns the new
186 * calibration value.
187 *
188 * This method assumes that the true sensor value is not actively changing at
189 * this time and computes an average from approximately 500 samples, 1 ms apart,
190 * for a 0.5 s period of calibration. The average value thus calculated is
191 * returned and stored for later calls to the adi_analog_read_calibrated() and
192 * adi_analog_read_calibrated_HR() functions. These functions will return
193 * the difference between this value and the current sensor value when called.
194 *
195 * Do not use this function when the sensor value might be unstable
196 * (gyro rotation, accelerometer movement).
197 *
198 * This function uses the following values of errno when an error state is
199 * reached:
200 * ENXIO - Either the ADI port value or the smart port value is not within its
201 * valid range (ADI port: 1-8, 'a'-'h', or 'A'-'H'; smart port: 1-21).
202 *
203 * \param smart_port
204 * The smart port number that the ADI Expander is in
205 * \param adi_port
206 * The ADI port to calibrate (from 1-8, 'a'-'h', 'A'-'H')
207 *
208 * \return The average sensor value computed by this function
209 *
210 * \b Example
211 * \code
212 *
213 * #define ADI_EXPANDER_PORT 20
214 * #define ANALOG_SENSOR_PORT 1
215 *
216 * void initialize() {
217 * ext_adi_analog_calibrate(ADI_EXPANDER_PORT, ANALOG_SENSOR_PORT);
218 * printf("Calibrated Reading: %d\n",
219 * ext_adi_analog_read_calibrated(ADI_EXPANDER_PORT, ANALOG_SENSOR_PORT));

302/**
303 * Gets the 16 bit calibrated value of an analog input port.
304 *
305 * The adi_analog_calibrate() function must be run first. This is intended for

320 *
321 * \param smart_port
322 * The smart port number that the ADI Expander is in
323 * \param adi_port
324 * The ADI port (from 1-8, 'a'-'h', 'A'-'H') for which the value will be
325 * returned
326 *
327 * \return The difference of the sensor value from its calibrated default from
328 * -16384 to 16384
329 *
330 * \b Example
331 * \code
332 *
333 * #define ADI_EXPANDER_PORT 20
334 * #define ANALOG_SENSOR_PORT 1
335 *
336 * void opcontrol() {
337 * while (true) {
338 * ext_adi_analog_calibrate(ADI_EXPANDER_PORT, ANALOG_SENSOR_PORT);
