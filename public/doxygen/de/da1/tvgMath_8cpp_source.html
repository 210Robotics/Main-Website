29bool mathInverse(const Matrix* m, Matrix* out)
30{
31 auto det = m->e11 * (m->e22 * m->e33 - m->e32 * m->e23) -
32 m->e12 * (m->e21 * m->e33 - m->e23 * m->e31) +
33 m->e13 * (m->e21 * m->e32 - m->e22 * m->e31);
34
35 if (mathZero(det)) return false;
36
37 auto invDet = 1 / det;
38
39 out->e11 = (m->e22 * m->e33 - m->e32 * m->e23) * invDet;
40 out->e12 = (m->e13 * m->e32 - m->e12 * m->e33) * invDet;
41 out->e13 = (m->e12 * m->e23 - m->e13 * m->e22) * invDet;
42 out->e21 = (m->e23 * m->e31 - m->e21 * m->e33) * invDet;
43 out->e22 = (m->e11 * m->e33 - m->e13 * m->e31) * invDet;
44 out->e23 = (m->e21 * m->e13 - m->e11 * m->e23) * invDet;
45 out->e31 = (m->e21 * m->e32 - m->e31 * m->e22) * invDet;
46 out->e32 = (m->e31 * m->e12 - m->e11 * m->e32) * invDet;
47 out->e33 = (m->e11 * m->e22 - m->e21 * m->e12) * invDet;
48
49 return true;
50}

53Matrix mathMultiply(const Matrix* lhs, const Matrix* rhs)
54{
55 Matrix m;
56
57 m.e11 = lhs->e11 * rhs->e11 + lhs->e12 * rhs->e21 + lhs->e13 * rhs->e31;
58 m.e12 = lhs->e11 * rhs->e12 + lhs->e12 * rhs->e22 + lhs->e13 * rhs->e32;
59 m.e13 = lhs->e11 * rhs->e13 + lhs->e12 * rhs->e23 + lhs->e13 * rhs->e33;
60
61 m.e21 = lhs->e21 * rhs->e11 + lhs->e22 * rhs->e21 + lhs->e23 * rhs->e31;
62 m.e22 = lhs->e21 * rhs->e12 + lhs->e22 * rhs->e22 + lhs->e23 * rhs->e32;
63 m.e23 = lhs->e21 * rhs->e13 + lhs->e22 * rhs->e23 + lhs->e23 * rhs->e33;
64
65 m.e31 = lhs->e31 * rhs->e11 + lhs->e32 * rhs->e21 + lhs->e33 * rhs->e31;
66 m.e32 = lhs->e31 * rhs->e12 + lhs->e32 * rhs->e22 + lhs->e33 * rhs->e32;
67 m.e33 = lhs->e31 * rhs->e13 + lhs->e32 * rhs->e23 + lhs->e33 * rhs->e33;
68
69 return m;
70}

73void mathRotate(Matrix* m, float degree)
74{
75 if (degree == 0.0f) return;
76
77 auto radian = degree / 180.0f * MATH_PI;
78 auto cosVal = cosf(radian);
79 auto sinVal = sinf(radian);
80
81 m->e12 = m->e11 * -sinVal;
82 m->e11 *= cosVal;
83 m->e21 = m->e22 * sinVal;
84 m->e22 *= cosVal;
85}

88bool mathIdentity(const Matrix* m)
89{
90 if (m->e11 != 1.0f || m->e12 != 0.0f || m->e13 != 0.0f ||
91 m->e21 != 0.0f || m->e22 != 1.0f || m->e23 != 0.0f ||
92 m->e31 != 0.0f || m->e32 != 0.0f || m->e33 != 1.0f) {
93 return false;
94 }
95 return true;
96}

99void mathMultiply(Point* pt, const Matrix* transform)
100{
101 auto tx = pt->x * transform->e11 + pt->y * transform->e12 + transform->e13;
102 auto ty = pt->x * transform->e21 + pt->y * transform->e22 + transform->e23;
103 pt->x = tx;
104 pt->y = ty;
105}
