33static inline __m128i ALPHA_BLEND(__m128i c, __m128i a)
34{
35 //1. set the masks for the A/G and R/B channels
36 auto AG = _mm_set1_epi32(0xff00ff00);
37 auto RB = _mm_set1_epi32(0x00ff00ff);
38
39 //2. mask the alpha vector - originally quartet [a, a, a, a]
40 auto aAG = _mm_and_si128(a, AG);
41 auto aRB = _mm_and_si128(a, RB);
42
43 //3. calculate the alpha blending of the 2nd and 4th channel
44 //- mask the color vector
45 //- multiply it by the masked alpha vector
46 //- add the correction to compensate bit shifting used instead of dividing by 255
47 //- shift bits - corresponding to division by 256
48 auto even = _mm_and_si128(c, RB);
49 even = _mm_mullo_epi16(even, aRB);
50 even =_mm_add_epi16(even, RB);
51 even = _mm_srli_epi16(even, 8);
52
53 //4. calculate the alpha blending of the 1st and 3rd channel:
54 //- mask the color vector
55 //- multiply it by the corresponding masked alpha vector and store the high bits of the result
56 //- add the correction to compensate division by 256 instead of by 255 (next step)
57 //- remove the low 8 bits to mimic the division by 256
58 auto odd = _mm_and_si128(c, AG);
59 odd = _mm_mulhi_epu16(odd, aAG);
60 odd = _mm_add_epi16(odd, RB);
61 odd = _mm_and_si128(odd, AG);
62
63 //5. the final result
64 return _mm_or_si128(odd, even);
65}

68static void avxRasterGrayscale8(uint8_t* dst, uint8_t val, uint32_t offset, int32_t len)
69{
70 dst += offset;
71
72 __m256i vecVal = _mm256_set1_epi8(val);
73
74 int32_t i = 0;
75 for (; i <= len - 32; i += 32) {
76 _mm256_storeu_si256((__m256i*)(dst + i), vecVal);
77 }
78
79 for (; i < len; ++i) {
80 dst[i] = val;
81 }
82}

85static void avxRasterPixel32(uint32_t *dst, uint32_t val, uint32_t offset, int32_t len)
86{
87 //1. calculate how many iterations we need to cover the length
88 uint32_t iterations = len / N_32BITS_IN_256REG;
89 uint32_t avxFilled = iterations * N_32BITS_IN_256REG;
90
91 //2. set the beginning of the array
92 dst += offset;
93
94 //3. fill the octets
95 for (uint32_t i = 0; i < iterations; ++i, dst += N_32BITS_IN_256REG) {
96 _mm256_storeu_si256((__m256i*)dst, _mm256_set1_epi32(val));
97 }
98
99 //4. fill leftovers (in the first step we have to set the pointer to the place where the avx job is done)
100 int32_t leftovers = len - avxFilled;
101 while (leftovers--) *dst++ = val;
102}

105static bool avxRasterTranslucentRect(SwSurface* surface, const SwBBox& region, uint8_t r, uint8_t g, uint8_t b, uint8_t a)
106{
107 if (surface->channelSize != sizeof(uint32_t)) {
108 TVGERR("SW_ENGINE", "Unsupported Channel Size = %d", surface->channelSize);
109 return false;
110 }
111
112 auto color = surface->join(r, g, b, a);
113 auto buffer = surface->buf32 + (region.min.y * surface->stride) + region.min.x;
114 auto h = static_cast<uint32_t>(region.max.y - region.min.y);
115 auto w = static_cast<uint32_t>(region.max.x - region.min.x);
116
117 uint32_t ialpha = 255 - a;
118
119 auto avxColor = _mm_set1_epi32(color);
120 auto avxIalpha = _mm_set1_epi8(ialpha);
121
122 for (uint32_t y = 0; y < h; ++y) {
123 auto dst = &buffer[y * surface->stride];
124
125 //1. fill the not aligned memory (for 128-bit registers a 16-bytes alignment is required)
126 auto notAligned = ((uintptr_t)dst & 0xf) / 4;
127 if (notAligned) {
128 notAligned = (N_32BITS_IN_128REG - notAligned > w ? w : N_32BITS_IN_128REG - notAligned);
129 for (uint32_t x = 0; x < notAligned; ++x, ++dst) {
130 *dst = color + ALPHA_BLEND(*dst, ialpha);
131 }
132 }
133
134 //2. fill the aligned memory - N_32BITS_IN_128REG pixels processed at once
135 uint32_t iterations = (w - notAligned) / N_32BITS_IN_128REG;
136 uint32_t avxFilled = iterations * N_32BITS_IN_128REG;
137 auto avxDst = (__m128i*)dst;
138 for (uint32_t x = 0; x < iterations; ++x, ++avxDst) {
139 *avxDst = _mm_add_epi32(avxColor, ALPHA_BLEND(*avxDst, avxIalpha));
140 }
141
142 //3. fill the remaining pixels
143 int32_t leftovers = w - notAligned - avxFilled;
144 dst += avxFilled;
145 while (leftovers--) {
146 *dst = color + ALPHA_BLEND(*dst, ialpha);
147 dst++;
148 }
149 }
150 return true;
151}

154static bool avxRasterTranslucentRle(SwSurface* surface, const SwRleData* rle, uint8_t r, uint8_t g, uint8_t b, uint8_t a)
155{
156 if (surface->channelSize != sizeof(uint32_t)) {
157 TVGERR("SW_ENGINE", "Unsupported Channel Size = %d", surface->channelSize);
158 return false;
159 }
160
161 auto color = surface->join(r, g, b, a);
162 auto span = rle->spans;
163 uint32_t src;
164
165 for (uint32_t i = 0; i < rle->size; ++i) {
166 auto dst = &surface->buf32[span->y * surface->stride + span->x];
167
168 if (span->coverage < 255) src = ALPHA_BLEND(color, span->coverage);
169 else src = color;
170
171 auto ialpha = IA(src);
172
173 //1. fill the not aligned memory (for 128-bit registers a 16-bytes alignment is required)
174 auto notAligned = ((uintptr_t)dst & 0xf) / 4;
175 if (notAligned) {
176 notAligned = (N_32BITS_IN_128REG - notAligned > span->len ? span->len : N_32BITS_IN_128REG - notAligned);
177 for (uint32_t x = 0; x < notAligned; ++x, ++dst) {
178 *dst = src + ALPHA_BLEND(*dst, ialpha);
179 }
180 }
181
182 //2. fill the aligned memory using avx - N_32BITS_IN_128REG pixels processed at once
183 //In order to avoid unnecessary avx variables declarations a check is made whether there are any iterations at all
184 uint32_t iterations = (span->len - notAligned) / N_32BITS_IN_128REG;
185 uint32_t avxFilled = 0;
186 if (iterations > 0) {
187 auto avxSrc = _mm_set1_epi32(src);
188 auto avxIalpha = _mm_set1_epi8(ialpha);
189
190 avxFilled = iterations * N_32BITS_IN_128REG;
191 auto avxDst = (__m128i*)dst;
192 for (uint32_t x = 0; x < iterations; ++x, ++avxDst) {
193 *avxDst = _mm_add_epi32(avxSrc, ALPHA_BLEND(*avxDst, avxIalpha));
194 }
195 }
196
197 //3. fill the remaining pixels
198 int32_t leftovers = span->len - notAligned - avxFilled;
199 dst += avxFilled;
200 while (leftovers--) {
201 *dst = src + ALPHA_BLEND(*dst, ialpha);
202 dst++;
203 }
204
205 ++span;
206 }
207 return true;
208}
