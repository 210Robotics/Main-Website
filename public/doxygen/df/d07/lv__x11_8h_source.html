53/**

56 * The minimal initialisation for initializing the X11 display driver with keyboard/mouse support:
57 * @code
58 * lv_display_t* disp = lv_x11_window_create("My Window Title", window_width, window_width);

60 * @endcode
61 * or with mouse cursor icon:
62 * @code
63 * lv_image_dsc_t mouse_symbol = {.....};

65 * lv_x11_inputs_create(disp, &mouse_symbol);
66 * @endcode
67 *
68 * @param[in] title title of the created X11 window
69 * @param[in] hor_res horizontal resolution (=width) of the X11 window
70 * @param[in] ver_res vertical resolution (=height) of the X11 window
71 * @return pointer to the display object
72 */
