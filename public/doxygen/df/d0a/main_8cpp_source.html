106/**
107 * Runs the user autonomous code. This function will be started in its own task
108 * with the default priority and stack size whenever the robot is enabled via
109 * the Field Management System or the VEX Competition Switch in the autonomous
110 * mode. Alternatively, this function may be called in initialize or opcontrol
111 * for non-competition testing purposes.
112 *
113 * If the robot is disabled or communications is lost, the autonomous task
114 * will be stopped. Re-enabling the robot will restart the task, not re-start it
115 * from where it left off.
116 */
117void autonomous() {
118 chassis.pid_targets_reset(); // Resets PID targets to 0
119 chassis.drive_imu_reset(); // Reset gyro position to 0
120 chassis.drive_sensor_reset(); // Reset drive sensors to 0
121 chassis.odom_xyt_set(0_in, 0_in, 0_deg); // Set the current position, you can start at a specific position with this
122 chassis.drive_brake_set(MOTOR_BRAKE_HOLD); // Set motors to hold. This helps autonomous consistency
123
124 /*
125 Odometry and Pure Pursuit are not magic
126
127 It is possible to get perfectly consistent results without tracking wheels,
128 but it is also possible to have extremely inconsistent results without tracking wheels.
129 When you don't use tracking wheels, you need to:
130 - avoid wheel slip
131 - avoid wheelies
132 - avoid throwing momentum around (super harsh turns, like in the example below)
133 You can do cool curved motions, but you have to give your robot the best chance
134 to be consistent
135 */
136
137 ez::as::auton_selector.selected_auton_call(); // Calls selected auton from autonomous selector
138}
