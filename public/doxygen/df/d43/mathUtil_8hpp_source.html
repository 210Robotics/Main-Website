115/**
116 * Integer power function. Computes `base^expo`.
117 *
118 * @param base The base.
119 * @param expo The exponent.
120 * @return `base^expo`.
121 */
122constexpr double ipow(const double base, const int expo) {
123 return (expo == 0) ? 1
124 : expo == 1 ? base
125 : expo > 1 ? ((expo & 1) ? base * ipow(base, expo - 1)
126 : ipow(base, expo / 2) * ipow(base, expo / 2))
127 : 1 / ipow(base, -expo);
128}

130/**
131 * Cuts out a range from the number. The new range of the input number will be
132 * `(-inf, min]U[max, +inf)`. If value sits equally between `min` and `max`, `max` will be returned.
133 *
134 * @param value The number to bound.
135 * @param min The lower bound of range.
136 * @param max The upper bound of range.
137 * @return The remapped value.
138 */
139constexpr double cutRange(const double value, const double min, const double max) {
140 const double middle = max - ((max - min) / 2);
141
142 if (value > min && value < middle) {
143 return min;
144 } else if (value <= max && value >= middle) {
145 return max;
146 }
147
148 return value;
149}

151/**
152 * Deadbands a range of the number. Returns the input value, or `0` if it is in the range `[min,
153 * max]`.
154 *
155 * @param value The number to deadband.
156 * @param min The lower bound of deadband.
157 * @param max The upper bound of deadband.
158 * @return The input value or `0` if it is in the range `[min, max]`.
159 */
160constexpr double deadband(const double value, const double min, const double max) {
161 return std::clamp(value, min, max) == value ? 0 : value;
162}

164/**
165 * Remap a value in the range `[oldMin, oldMax]` to the range `[newMin, newMax]`.
166 *
167 * @param value The value in the old range.
168 * @param oldMin The old range lower bound.
169 * @param oldMax The old range upper bound.
170 * @param newMin The new range lower bound.
171 * @param newMax The new range upper bound.
172 * @return The input value in the new range `[newMin, newMax]`.
173 */
174constexpr double remapRange(const double value,
175 const double oldMin,
176 const double oldMax,
177 const double newMin,
178 const double newMax) {
179 return (value - oldMin) * ((newMax - newMin) / (oldMax - oldMin)) + newMin;
180}

233/**
234 * Maps ADI port numbers/chars to numbers:
235 * ```
236 * when (port) {
237 * in ['a', 'h'] -> [1, 8]
238 * in ['A', 'H'] -> [1, 8]
239 * else -> [1, 8]
240 * }
241 * ```
242 *
243 * @param port The ADI port number or char.
244 * @return An equivalent ADI port number.
245 */
246constexpr std::int8_t transformADIPort(const std::int8_t port) {
247 if (port >= 'a' && port <= 'h') {
248 return port - ('a' - 1);
249 } else if (port >= 'A' && port <= 'H') {
250 return port - ('A' - 1);
251 } else {
252 return port;
253 }
254}
