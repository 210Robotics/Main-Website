15 /**
16 * Model for an h-drive (wheels parallel with robot's direction of motion, with an additional
17 * wheel perpendicular to those). When the left and right side motors are powered +100%, the robot
18 * should move forward in a straight line. When the middle motor is powered +100%, the robot
19 * should strafe right in a straight line.
20 *
21 * @param ileftSideMotor The left side motor.
22 * @param irightSideMotor The right side motor.
23 * @param imiddleMotor The middle (perpendicular) motor.
24 * @param ileftEnc The left side encoder.
25 * @param irightEnc The right side encoder.
26 * @param imiddleEnc The middle encoder.
27 */
28 HDriveModel(std::shared_ptr<AbstractMotor> ileftSideMotor,
29 std::shared_ptr<AbstractMotor> irightSideMotor,
30 std::shared_ptr<AbstractMotor> imiddleMotor,
31 std::shared_ptr<ContinuousRotarySensor> ileftEnc,
32 std::shared_ptr<ContinuousRotarySensor> irightEnc,
33 std::shared_ptr<ContinuousRotarySensor> imiddleEnc,
34 double imaxVelocity,
35 double imaxVoltage);

37 /**
38 * Drive the robot forwards (using open-loop control). Uses velocity mode. Sets the middle motor
39 * to zero velocity.
40 *
41 * @param ispeed motor power

43 void forward(double ispeed) override;
44
45 /**
46 * Drive the robot in an arc (using open-loop control). Uses velocity mode. Sets the middle motor
47 * to zero velocity.
48 *
49 * The algorithm is (approximately):
50 * leftPower = ySpeed + zRotation
51 * rightPower = ySpeed - zRotation
52 *
53 * @param iySpeed speed on y axis (forward)
54 * @param izRotation speed around z axis (up)
55 */
56 void driveVector(double iySpeed, double izRotation) override;
57
58 /**
59 * Drive the robot in an arc. Uses voltage mode. Sets the middle motor to zero velocity.
60 *
61 * The algorithm is (approximately):
62 * leftPower = forwardSpeed + yaw
63 * rightPower = forwardSpeed - yaw
64 *
65 * @param iforwadSpeed speed in the forward direction
66 * @param iyaw speed around the vertical axis
67 */
68 void driveVectorVoltage(double iforwardSpeed, double iyaw) override;
69
70 /**
71 * Turn the robot clockwise (using open-loop control). Uses velocity mode. Sets the middle motor
72 * to zero velocity.
73 *
74 * @param ispeed motor power

76 void rotate(double ispeed) override;
77
78 /**

81 void stop() override;

83 /**
84 * Drive the robot with a tank drive layout. Uses voltage mode. Sets the middle motor to zero
85 * velocity.
86 *
87 * @param ileftSpeed left side speed
88 * @param irightSpeed right side speed
89 * @param ithreshold deadband on joystick values
90 */
91 void tank(double ileftSpeed, double irightSpeed, double ithreshold = 0) override;
92
93 /**
94 * Drive the robot with an arcade drive layout. Uses voltage mode. Sets the middle motor to zero
95 * velocity.
96 *
97 * @param iforwardSpeed speed in the forward direction
98 * @param iyaw speed around the vertical axis
99 * @param ithreshold deadband on joystick values
100 */
101 void arcade(double iforwardSpeed, double iyaw, double ithreshold = 0) override;
102
103 /**
104 * Drive the robot with a curvature drive layout. The robot drives in constant radius turns
105 * where you control the curvature (inverse of radius) you drive in. This is advantageous
106 * because the forward speed will not affect the rate of turning. The algorithm switches to
107 * arcade if the forward speed is 0. Uses voltage mode. Sets the middle motor to zero velocity.
108 *
109 * @param iforwardSpeed speed in the forward direction
110 * @param icurvature curvature (inverse of radius) to drive in
111 * @param ithreshold deadband on joystick values
112 */
113 void curvature(double iforwardSpeed, double icurvature, double ithreshold = 0) override;
114
115 /**
116 * Drive the robot with an arcade drive layout. Uses voltage mode.
117 *
118 * @param irightSpeed speed to the right
119 * @param iforwardSpeed speed in the forward direction
120 * @param iyaw speed around the vertical axis
121 * @param ithreshold deadband on joystick values
122 */
123 virtual void
124 hArcade(double irightSpeed, double iforwardSpeed, double iyaw, double ithreshold = 0);

126 /**
127 * Drive the robot with an curvature drive layout. Uses voltage mode.
128 *
129 * @param irightSpeed speed to the right
130 * @param iforwardSpeed speed in the forward direction
131 * @param icurvature curvature (inverse of radius) to drive in
132 * @param ithreshold deadband on joystick values
133 */
134 virtual void
135 hCurvature(double irightSpeed, double iforwardSpeed, double icurvature, double ithreshold = 0);
